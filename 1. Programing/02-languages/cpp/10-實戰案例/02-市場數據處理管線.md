# 市場數據處理管線

## 概述

市場數據 (Market Data) 處理是高頻交易系統的核心組件。本文涵蓋從原始數據接收、解析、驗證到分發的完整管線設計。

### 數據特性

| 特性 | 說明 | 典型值 |
|-----|------|-------|
| 數據速率 | 每秒訊息數 | 100K - 10M msg/s |
| 延遲要求 | 接收到處理完成 | < 10 μs |
| 數據類型 | Quote, Trade, Depth | 多種格式 |
| 順序性 | 必須保持序列順序 | 嚴格要求 |

### 架構概覽

```
┌─────────────┐   ┌──────────┐   ┌─────────┐   ┌──────────┐
│ Network RX  │──>│  Parser  │──>│ Norma-  │──>│ Dispatch │
```

## 原始數據接收

### UDP 多播接收

```cpp
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <cstring>
#include <stdexcept>

class MulticastReceiver {
    int sockfd_;
    sockaddr_in local_addr_;
    
public:
    MulticastReceiver(const char* group_ip, uint16_t port, const char* interface_ip = nullptr) {
        // 創建 UDP socket
        sockfd_ = socket(AF_INET, SOCK_DGRAM, 0);
        if (sockfd_ < 0) {
            throw std::runtime_error("socket creation failed");
        }
        
        // 允許地址重用
        int reuse = 1;
        setsockopt(sockfd_, SOL_SOCKET, SO_REUSEADDR, &reuse, sizeof(reuse));
        
        // 綁定本地地址
        memset(&local_addr_, 0, sizeof(local_addr_));
        local_addr_.sin_family = AF_INET;
        local_addr_.sin_port = htons(port);
        local_addr_.sin_addr.s_addr = INADDR_ANY;
        
        if (bind(sockfd_, (sockaddr*)&local_addr_, sizeof(local_addr_)) < 0) {
            throw std::runtime_error("bind failed");
        }
        
        // 加入多播組
        ip_mreq mreq;
        mreq.imr_multiaddr.s_addr = inet_addr(group_ip);
        
        if (interface_ip) {
            mreq.imr_interface.s_addr = inet_addr(interface_ip);
        } else {
            mreq.imr_interface.s_addr = INADDR_ANY;
        }
        
        if (setsockopt(sockfd_, IPPROTO_IP, IP_ADD_MEMBERSHIP, 
                      &mreq, sizeof(mreq)) < 0) {
            throw std::runtime_error("join multicast group failed");
        }
        
        // 設定接收緩衝區 (8 MB)
        int buffer_size = 8 * 1024 * 1024;
        setsockopt(sockfd_, SOL_SOCKET, SO_RCVBUF, &buffer_size, sizeof(buffer_size));
    }
    
    ~MulticastReceiver() {
        if (sockfd_ >= 0) {
            close(sockfd_);
        }
    }
    
    ssize_t receive(void* buffer, size_t len) {
        return recvfrom(sockfd_, buffer, len, 0, nullptr, nullptr);
    }
    
    int socket_fd() const { return sockfd_; }
};

// 使用範例
void receive_market_data() {
    MulticastReceiver receiver("239.1.1.1", 10000, "192.168.1.100");
    
    uint8_t buffer[65536];
    
    while (true) {
        ssize_t bytes_received = receiver.receive(buffer, sizeof(buffer));
        
        if (bytes_received > 0) {
            // 處理數據包
            process_packet(buffer, bytes_received);
        }
    }
}
```

### 零拷貝接收 (recvmmsg)

```cpp
#include <sys/socket.h>
#include <vector>

class BatchReceiver {
    int sockfd_;
    static constexpr size_t BATCH_SIZE = 32;
    static constexpr size_t BUFFER_SIZE = 2048;
    
    struct MessageBuffer {
        uint8_t data[BUFFER_SIZE];
        iovec iov;
        mmsghdr msg;
    };
    
    std::vector<MessageBuffer> buffers_;
    
public:
    explicit BatchReceiver(int sockfd) : sockfd_(sockfd) {
        buffers_.resize(BATCH_SIZE);
        
        for (auto& buf : buffers_) {
            buf.iov.iov_base = buf.data;
            buf.iov.iov_len = BUFFER_SIZE;
            
            memset(&buf.msg, 0, sizeof(buf.msg));
            buf.msg.msg_hdr.msg_iov = &buf.iov;
            buf.msg.msg_hdr.msg_iovlen = 1;
        }
    }
    
    int receive_batch() {
        return recvmmsg(sockfd_, buffers_.data(), BATCH_SIZE, 
                       MSG_WAITFORONE, nullptr);
    }
    
    const uint8_t* get_message(int index, size_t& len) const {
        len = buffers_[index].msg.msg_len;
        return buffers_[index].data;
    }
};

// 使用範例
void batch_receive_loop(int sockfd) {
    BatchReceiver receiver(sockfd);
    
    while (true) {
        int count = receiver.receive_batch();
        
        for (int i = 0; i < count; ++i) {
            size_t len;
            const uint8_t* data = receiver.get_message(i, len);
            
            process_packet(data, len);
        }
    }
}
```

```cpp
#include <cstdint>
#include <cstring>
#include <endian.h>

// 市場數據訊息格式
#pragma pack(push, 1)
struct MarketDataHeader {
    uint16_t message_length;
    uint16_t message_type;
    uint32_t sequence_number;
    uint64_t timestamp;
};

struct QuoteMessage {
    MarketDataHeader header;
    uint32_t symbol_id;
    uint64_t bid_price;    // 固定點數 (x 10000)
    uint64_t ask_price;
    uint64_t bid_size;
    uint64_t ask_size;
};

struct TradeMessage {
    MarketDataHeader header;
    uint32_t symbol_id;
    uint64_t price;
    uint64_t volume;
    uint8_t side;  // 'B' or 'S'
    uint8_t padding[3];
};
#pragma pack(pop)

class BinaryParser {
public:
    enum class MessageType : uint16_t {
        QUOTE = 1,
        TRADE = 2,
        DEPTH = 3
    };
    
    static bool parse_header(const uint8_t* data, size_t len, MarketDataHeader& header) {
        if (len < sizeof(MarketDataHeader)) {
            return false;
        }
        
        memcpy(&header, data, sizeof(MarketDataHeader));
        
        // 網路字節序轉換
        header.message_length = be16toh(header.message_length);
        header.message_type = be16toh(header.message_type);
        header.sequence_number = be32toh(header.sequence_number);
        header.timestamp = be64toh(header.timestamp);
        
        return true;
    }
    
    static bool parse_quote(const uint8_t* data, size_t len, QuoteMessage& quote) {
        if (len < sizeof(QuoteMessage)) {
            return false;
        }
        
        memcpy(&quote, data, sizeof(QuoteMessage));
        
        parse_header(data, len, quote.header);
        
        quote.symbol_id = be32toh(quote.symbol_id);
        quote.bid_price = be64toh(quote.bid_price);
        quote.ask_price = be64toh(quote.ask_price);
        quote.bid_size = be64toh(quote.bid_size);
        quote.ask_size = be64toh(quote.ask_size);
        
        return true;
    }
    
    static bool parse_trade(const uint8_t* data, size_t len, TradeMessage& trade) {
        if (len < sizeof(TradeMessage)) {
            return false;
        }
        
        memcpy(&trade, data, sizeof(TradeMessage));
        
        parse_header(data, len, trade.header);
        
        trade.symbol_id = be32toh(trade.symbol_id);
        trade.price = be64toh(trade.price);
        trade.volume = be64toh(trade.volume);
        
        return true;
    }
};
```

### 文本協議解析 (FIX-like)

```cpp
#include <string_view>
#include <unordered_map>

class TextParser {
public:
    using FieldMap = std::unordered_map<int, std::string_view>;
    
    static bool parse_message(std::string_view message, FieldMap& fields) {
        fields.clear();
        
        size_t pos = 0;
        
        while (pos < message.size()) {
            // 查找下一個分隔符 (SOH = 0x01)
            size_t delim = message.find('\x01', pos);
            if (delim == std::string_view::npos) {
                delim = message.size();
            }
            
            std::string_view field = message.substr(pos, delim - pos);
            
            // 解析 tag=value
            size_t eq = field.find('=');
            if (eq != std::string_view::npos) {
                int tag = parse_int(field.substr(0, eq));
                std::string_view value = field.substr(eq + 1);
                
                fields[tag] = value;
            }
            
            pos = delim + 1;
        }
        
        return !fields.empty();
    }
    
    static int parse_int(std::string_view str) {
        int result = 0;
        
        for (char c : str) {
            if (c >= '0' && c <= '9') {
                result = result * 10 + (c - '0');
            }
        }
        
        return result;
    }
    
    static double parse_double(std::string_view str) {
        // 簡化實現，實際應使用 fast_float
        char buffer[32];
        size_t len = std::min(str.size(), sizeof(buffer) - 1);
        memcpy(buffer, str.data(), len);
        buffer[len] = '\0';
        
        return std::atof(buffer);
    }
};
```

## 序列號管理

### 序列號檢測器

```cpp
#include <atomic>
#include <vector>

class SequenceChecker {
    std::atomic<uint32_t> expected_seq_;
    std::atomic<uint64_t> gap_count_{0};
    std::atomic<uint64_t> duplicate_count_{0};
    
public:
    explicit SequenceChecker(uint32_t initial_seq = 1)
        : expected_seq_(initial_seq) {}
    
    enum class Status {
        IN_SEQUENCE,
        GAP_DETECTED,
        DUPLICATE
    };
    
    Status check(uint32_t seq_num) {
        uint32_t expected = expected_seq_.load(std::memory_order_relaxed);
        
        if (seq_num == expected) {
            expected_seq_.fetch_add(1, std::memory_order_relaxed);
            return Status::IN_SEQUENCE;
        } else if (seq_num > expected) {
            // 序列號跳躍 (gap)
            gap_count_.fetch_add(1, std::memory_order_relaxed);
            expected_seq_.store(seq_num + 1, std::memory_order_relaxed);
            return Status::GAP_DETECTED;
        } else {
            // 重複或亂序
            duplicate_count_.fetch_add(1, std::memory_order_relaxed);
            return Status::DUPLICATE;
        }
    }
    
    void reset(uint32_t seq) {
        expected_seq_.store(seq, std::memory_order_relaxed);
    }
    
    uint64_t gap_count() const {
        return gap_count_.load(std::memory_order_relaxed);
    }
    
    uint64_t duplicate_count() const {
        return duplicate_count_.load(std::memory_order_relaxed);
    }
};

// 多頻道序列號管理
class MultiChannelSequenceChecker {
    std::vector<SequenceChecker> checkers_;
    
public:
    explicit MultiChannelSequenceChecker(size_t num_channels) {
        checkers_.reserve(num_channels);
        for (size_t i = 0; i < num_channels; ++i) {
            checkers_.emplace_back(1);
        }
    }
    
    SequenceChecker::Status check(size_t channel, uint32_t seq_num) {
        if (channel >= checkers_.size()) {
            return SequenceChecker::Status::DUPLICATE;
        }
        
        return checkers_[channel].check(seq_num);
    }
    
    void print_stats() const {
        for (size_t i = 0; i < checkers_.size(); ++i) {
            std::cout << "Channel " << i 
                      << " - Gaps: " << checkers_[i].gap_count()
                      << ", Duplicates: " << checkers_[i].duplicate_count() << "\n";
        }
    }
};
```

## 數據標準化

### 價格標準化

```cpp
#include <cstdint>

class PriceNormalizer {
    double tick_size_;
    uint64_t price_scale_;
    
public:
    explicit PriceNormalizer(double tick_size)
        : tick_size_(tick_size) {
        
        // 計算縮放因子 (例如 0.01 -> 10000)
        price_scale_ = static_cast<uint64_t>(1.0 / tick_size_);
    }
    
    // 原始價格 -> 標準化價格 (整數)
    uint64_t normalize(double price) const {
        return static_cast<uint64_t>(price * price_scale_ + 0.5);
    }
    
    // 標準化價格 -> 原始價格 (浮點)
    double denormalize(uint64_t price) const {
        return static_cast<double>(price) / price_scale_;
    }
    
    // 固定點數格式轉換 (交易所常用)
    uint64_t from_fixed_point(uint64_t fixed_price, uint32_t decimals) const {
        double divisor = 1.0;
        for (uint32_t i = 0; i < decimals; ++i) {
            divisor *= 10.0;
        }
        
        double price = static_cast<double>(fixed_price) / divisor;
        return normalize(price);
    }
};

// 多商品價格標準化
class MultiSymbolPriceNormalizer {
    std::unordered_map<uint32_t, PriceNormalizer> normalizers_;
    
public:
    void add_symbol(uint32_t symbol_id, double tick_size) {
        normalizers_.emplace(symbol_id, PriceNormalizer(tick_size));
    }
    
    uint64_t normalize(uint32_t symbol_id, double price) const {
        auto it = normalizers_.find(symbol_id);
        if (it != normalizers_.end()) {
            return it->second.normalize(price);
        }
        return 0;
    }
    
    double denormalize(uint32_t symbol_id, uint64_t price) const {
        auto it = normalizers_.find(symbol_id);
        if (it != normalizers_.end()) {
            return it->second.denormalize(price);
        }
        return 0.0;
    }
};
```

### 主題訂閱模式

```cpp
#include <unordered_map>
#include <vector>
#include <functional>
#include <mutex>

template<typename T>
class MessageDispatcher {
public:
    using Callback = std::function<void(const T&)>;
    using SubscriptionId = uint64_t;
    
private:
    struct Subscription {
        SubscriptionId id;
        Callback callback;
    };
    
    std::unordered_map<uint32_t, std::vector<Subscription>> subscribers_;
    std::mutex mutex_;
    SubscriptionId next_id_{1};
    
public:
    SubscriptionId subscribe(uint32_t symbol_id, Callback callback) {
        std::lock_guard<std::mutex> lock(mutex_);
        
        SubscriptionId id = next_id_++;
        
        subscribers_[symbol_id].push_back(Subscription{id, std::move(callback)});
        
        return id;
    }
    
    void unsubscribe(uint32_t symbol_id, SubscriptionId id) {
        std::lock_guard<std::mutex> lock(mutex_);
        
        auto it = subscribers_.find(symbol_id);
        if (it != subscribers_.end()) {
            auto& subs = it->second;
            
            subs.erase(
                std::remove_if(subs.begin(), subs.end(),
                    [id](const Subscription& sub) { return sub.id == id; }),
                subs.end()
            );
        }
    }
    
    void dispatch(uint32_t symbol_id, const T& message) {
        std::vector<Callback> callbacks;
        
        {
            std::lock_guard<std::mutex> lock(mutex_);
            
            auto it = subscribers_.find(symbol_id);
            if (it != subscribers_.end()) {
                callbacks.reserve(it->second.size());
                
                for (const auto& sub : it->second) {
                    callbacks.push_back(sub.callback);
                }
            }
        }
        
        // 在鎖外執行回調
        for (const auto& callback : callbacks) {
            callback(message);
        }
    }
};

// 使用範例
struct NormalizedQuote {

    uint64_t bid_price;

    uint64_t bid_size;

};

    // 策略處理邏輯
}
```

### 無鎖分發器

```cpp
#include <atomic>
#include <array>

template<typename T, size_t MaxSubscribers = 16>
class LockFreeDispatcher {
    struct Subscriber {
        std::atomic<bool> active{false};
        void (*callback)(const T&);
        alignas(64) char padding[64 - sizeof(std::atomic<bool>) - sizeof(void*)];
    };
    
    std::array<Subscriber, MaxSubscribers> subscribers_;
    
public:
    bool subscribe(void (*callback)(const T&)) {
        for (auto& sub : subscribers_) {
            bool expected = false;
            if (sub.active.compare_exchange_strong(expected, true)) {
                sub.callback = callback;
                return true;
            }
        }
        
        return false;  // 已滿
    }
    
    void dispatch(const T& message) {
        for (auto& sub : subscribers_) {
            if (sub.active.load(std::memory_order_acquire)) {
                sub.callback(message);
            }
        }
    }
};
```

## 完整處理管線

### 端到端實現

```cpp
#include <thread>
#include <atomic>

class MarketDataPipeline {
    // 網路接收
    MulticastReceiver receiver_;
    
    // 解析執行緒
    std::thread parser_thread_;
    
    // 序列號檢查
    SequenceChecker seq_checker_;
    
    // 價格標準化
    MultiSymbolPriceNormalizer price_normalizer_;
    
    // 訊息分發
    MessageDispatcher<NormalizedQuote> dispatcher_;
    
    // 統計
    alignas(64) std::atomic<uint64_t> messages_received_{0};
    alignas(64) std::atomic<uint64_t> messages_processed_{0};
    alignas(64) std::atomic<uint64_t> parsing_errors_{0};
    
    std::atomic<bool> running_{false};
    
public:
    MarketDataPipeline(const char* mcast_group, uint16_t port)
        : receiver_(mcast_group, port), seq_checker_(1) {
        
        // 配置價格標準化器
        price_normalizer_.add_symbol(1001, 0.01);  // AAPL
        price_normalizer_.add_symbol(1002, 0.01);  // GOOGL
    }
    
    void start() {
        running_.store(true, std::memory_order_relaxed);
        
        parser_thread_ = std::thread(&MarketDataPipeline::parse_loop, this);
        
        // 設定執行緒親和性
        cpu_set_t cpuset;
        CPU_ZERO(&cpuset);
        CPU_SET(2, &cpuset);
        pthread_setaffinity_np(parser_thread_.native_handle(), 
                              sizeof(cpu_set_t), &cpuset);
    }
    
    void stop() {
        running_.store(false, std::memory_order_relaxed);
        
        if (parser_thread_.joinable()) {
            parser_thread_.join();
        }
    }
    
    void subscribe_quotes(uint32_t symbol_id, 
                         std::function<void(const NormalizedQuote&)> callback) {
        dispatcher_.subscribe(symbol_id, callback);
    }
    
    void print_stats() const {
        uint64_t received = messages_received_.load(std::memory_order_relaxed);
        uint64_t processed = messages_processed_.load(std::memory_order_relaxed);
        uint64_t errors = parsing_errors_.load(std::memory_order_relaxed);
        
        std::cout << "Received: " << received
                  << ", Processed: " << processed
                  << ", Errors: " << errors
                  << ", Gaps: " << seq_checker_.gap_count()
                  << ", Duplicates: " << seq_checker_.duplicate_count() << "\n";
    }
    
private:
    void parse_loop() {
        uint8_t buffer[65536];
        
        while (running_.load(std::memory_order_relaxed)) {
            ssize_t bytes = receiver_.receive(buffer, sizeof(buffer));
            
            if (bytes > 0) {
                messages_received_.fetch_add(1, std::memory_order_relaxed);
                
                process_packet(buffer, bytes);
            }
        }
    }
    
    void process_packet(const uint8_t* data, size_t len) {
        MarketDataHeader header;
        
        if (!BinaryParser::parse_header(data, len, header)) {
            parsing_errors_.fetch_add(1, std::memory_order_relaxed);
            return;
        }
        
        // 檢查序列號
        auto seq_status = seq_checker_.check(header.sequence_number);
        
        if (seq_status == SequenceChecker::Status::GAP_DETECTED) {
            // 記錄 gap，可能觸發重傳請求
        }
        
        // 根據訊息類型解析
        if (header.message_type == static_cast<uint16_t>(BinaryParser::MessageType::QUOTE)) {
            QuoteMessage quote;
            
            if (BinaryParser::parse_quote(data, len, quote)) {
                process_quote(quote);
            } else {
                parsing_errors_.fetch_add(1, std::memory_order_relaxed);
            }
        }
    }
    
    void process_quote(const QuoteMessage& quote) {
        // 標準化價格
        NormalizedQuote normalized;
        normalized.symbol_id = quote.symbol_id;
        normalized.bid_price = price_normalizer_.normalize(
            quote.symbol_id, 
            static_cast<double>(quote.bid_price) / 10000.0
        );
        normalized.ask_price = price_normalizer_.normalize(
            quote.symbol_id,
            static_cast<double>(quote.ask_price) / 10000.0
        );
        normalized.bid_size = quote.bid_size;
        normalized.ask_size = quote.ask_size;
        normalized.timestamp = quote.header.timestamp;
        
        // 分發
        dispatcher_.dispatch(quote.symbol_id, normalized);
        
        messages_processed_.fetch_add(1, std::memory_order_relaxed);
    }
};
```

```cpp
template<typename T, size_t BatchSize>
class BatchProcessor {
    std::array<T, BatchSize> batch_;
    size_t count_{0};
    
    std::function<void(const T*, size_t)> batch_handler_;
    
public:
    explicit BatchProcessor(std::function<void(const T*, size_t)> handler)
        : batch_handler_(std::move(handler)) {}
    
    void add(const T& item) {
        batch_[count_++] = item;
        
        if (count_ >= BatchSize) {
            flush();
        }
    }
    
    void flush() {
        if (count_ > 0) {
            batch_handler_(batch_.data(), count_);
            count_ = 0;
        }
    }
};
```

```cpp
#include <immintrin.h>

// SIMD 加速的整數解析
class SIMDParser {
public:
    // 解析 8 個連續的 uint32_t (大端序)
    static void parse_uint32_batch(const uint8_t* data, uint32_t* output) {
        // 加載 32 字節 (8 個 uint32_t)
        __m256i values = _mm256_loadu_si256(reinterpret_cast<const __m256i*>(data));
        
        // 字節序轉換 (大端 -> 小端)
        __m256i shuffled = _mm256_shuffle_epi8(values,
            _mm256_set_epi8(
                12, 13, 14, 15,  8,  9, 10, 11,  4,  5,  6,  7,  0,  1,  2,  3,
                12, 13, 14, 15,  8,  9, 10, 11,  4,  5,  6,  7,  0,  1,  2,  3
            )
        );
        
        // 存儲
        _mm256_storeu_si256(reinterpret_cast<__m256i*>(output), shuffled);
```

## 延遲測量

### 端到端延遲追蹤

```cpp
#include <chrono>

struct LatencyTracker {
    uint64_t network_rx_tsc;
    uint64_t parse_complete_tsc;
    uint64_t normalize_complete_tsc;
    uint64_t dispatch_complete_tsc;
    
    void mark_network_rx() {
        network_rx_tsc = __rdtsc();
    }
    
    void mark_parse_complete() {
        parse_complete_tsc = __rdtsc();
    }
    
    void mark_normalize_complete() {
        normalize_complete_tsc = __rdtsc();
    }
    
    void mark_dispatch_complete() {
        dispatch_complete_tsc = __rdtsc();
    }
    
    void print_breakdown(uint64_t cpu_freq_mhz) const {
        auto to_ns = [cpu_freq_mhz](uint64_t cycles) {
            return (cycles * 1000) / cpu_freq_mhz;
        };
        
        std::cout << "Latency breakdown:\n"
                  << "  Parse:     " << to_ns(parse_complete_tsc - network_rx_tsc) << " ns\n"
                  << "  Normalize: " << to_ns(normalize_complete_tsc - parse_complete_tsc) << " ns\n"
                  << "  Dispatch:  " << to_ns(dispatch_complete_tsc - normalize_complete_tsc) << " ns\n"
                  << "  Total:     " << to_ns(dispatch_complete_tsc - network_rx_tsc) << " ns\n";
    }
};
```
