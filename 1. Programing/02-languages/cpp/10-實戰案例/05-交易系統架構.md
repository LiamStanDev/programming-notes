# 交易系統架構

## 概述

本文涵蓋完整交易系統的架構設計，包含市場數據處理、訂單管理、風險控制與執行引擎的整合。

### 系統組件

```
┌─────────────────────────────────────────────────────────┐
│                    Trading System                        │
├──────────────┬────────────────┬─────────────────────────┤
│ Market Data  │  Risk Manager  │   Order Management      │
│   Handler    │                │       System            │
├──────────────┼────────────────┼─────────────────────────┤
│              │                │                         │
│  Orderbook   │  Position Mgr  │    Execution Engine     │
│              │                │                         │
└──────────────┴────────────────┴─────────────────────────┘
        │                │                    │
        ↓                ↓                    ↓
   Strategies    Risk Checks          Exchange Gateway
```

## 核心架構

### 主要組件

```cpp
#include <memory>
#include <thread>
#include <atomic>

class TradingSystem {
public:
    struct Config {
        std::string market_data_host;
        uint16_t market_data_port;
        std::string order_gateway_host;
        uint16_t order_gateway_port;
        double max_position_value;
        double max_order_value;
    };
    
private:
    Config config_;
    
    // 核心組件
    std::unique_ptr<MarketDataHandler> market_data_;
    std::unique_ptr<OrderManagementSystem> oms_;
    std::unique_ptr<RiskManager> risk_manager_;
    std::unique_ptr<ExecutionEngine> execution_;
    
    // 執行緒
    std::vector<std::thread> threads_;
    std::atomic<bool> running_{false};
    
public:
    explicit TradingSystem(Config config) : config_(std::move(config)) {
        initialize();
    }
    
    void start() {
        running_.store(true);
        
        // 啟動各組件
        market_data_->start();
        oms_->start();
        execution_->start();
        
        // 啟動監控執行緒
        threads_.emplace_back(&TradingSystem::monitoring_loop, this);
    }
    
    void stop() {
        running_.store(false);
        
        market_data_->stop();
        oms_->stop();
        execution_->stop();
        
        for (auto& t : threads_) {
            if (t.joinable()) {
                t.join();
            }
        }
    }
    
    // 策略介面
    void register_strategy(std::shared_ptr<TradingStrategy> strategy) {
        // 連接市場數據
        market_data_->subscribe(strategy->symbols(), 
            [strategy](const MarketData& data) {
                strategy->on_market_data(data);
            });
        
        // 連接訂單回報
        oms_->register_callback(
            [strategy](const OrderUpdate& update) {
                strategy->on_order_update(update);
            });
    }
    
private:
    void initialize() {
        market_data_ = std::make_unique<MarketDataHandler>(
            config_.market_data_host,
            config_.market_data_port
        );
        
        oms_ = std::make_unique<OrderManagementSystem>();
        
        risk_manager_ = std::make_unique<RiskManager>(
            config_.max_position_value,
            config_.max_order_value
        );
        
        execution_ = std::make_unique<ExecutionEngine>(
            config_.order_gateway_host,
            config_.order_gateway_port
        );
    }
    
    void monitoring_loop() {
        while (running_.load()) {
            std::this_thread::sleep_for(std::chrono::seconds(1));
            
            // 監控系統狀態
            log_system_status();
        }
    }
    
    void log_system_status() {
        // 記錄各組件狀態、延遲、吞吐量等
    }
};
```

## 策略框架

### 策略基類

```cpp
class TradingStrategy {
public:
    virtual ~TradingStrategy() = default;
    
    // 策略生命週期
    virtual void on_start() = 0;
    virtual void on_stop() = 0;
    
    // 市場數據回調
    virtual void on_market_data(const MarketData& data) = 0;
    virtual void on_trade(const Trade& trade) = 0;
    
    // 訂單回調
    virtual void on_order_update(const OrderUpdate& update) = 0;
    virtual void on_fill(const Fill& fill) = 0;
    
    // 配置
    virtual std::vector<std::string> symbols() const = 0;
    virtual std::string name() const = 0;
};

// 簡單做市策略範例
class SimpleMarketMaker : public TradingStrategy {
    std::string symbol_;
    double spread_;
    uint64_t quote_size_;
    
    OrderManagementSystem& oms_;
    
    double last_mid_price_{0.0};
    uint64_t bid_order_id_{0};
    uint64_t ask_order_id_{0};
    
public:
    SimpleMarketMaker(std::string symbol, double spread, uint64_t size,
                      OrderManagementSystem& oms)
        : symbol_(std::move(symbol)), spread_(spread), 
          quote_size_(size), oms_(oms) {}
    
    void on_start() override {
        std::cout << "Strategy started: " << name() << "\n";
    }
    
    void on_stop() override {
        cancel_all_orders();
    }
    
    void on_market_data(const MarketData& data) override {
        if (data.symbol != symbol_) return;
        
        double mid_price = (data.bid_price + data.ask_price) / 2.0;
        
        if (std::abs(mid_price - last_mid_price_) > 0.01) {
            update_quotes(mid_price);
            last_mid_price_ = mid_price;
        }
    }
    
    void on_trade(const Trade& trade) override {}
    
    void on_order_update(const OrderUpdate& update) override {
        if (update.status == OrderStatus::FILLED) {
            // 訂單成交，重新報價
            update_quotes(last_mid_price_);
        }
    }
    
    void on_fill(const Fill& fill) override {}
    
    std::vector<std::string> symbols() const override {
        return {symbol_};
    }
    
    std::string name() const override {
        return "SimpleMarketMaker_" + symbol_;
    }
    
private:
    void update_quotes(double mid_price) {
        cancel_all_orders();
        
        double bid_price = mid_price - spread_ / 2.0;
        double ask_price = mid_price + spread_ / 2.0;
        
        bid_order_id_ = oms_.submit_order(
            symbol_, Side::BUY, bid_price, quote_size_
        );
        
        ask_order_id_ = oms_.submit_order(
            symbol_, Side::SELL, ask_price, quote_size_
        );
    }
    
    void cancel_all_orders() {
        if (bid_order_id_ != 0) {
            oms_.cancel_order(bid_order_id_);
            bid_order_id_ = 0;
        }
        
        if (ask_order_id_ != 0) {
            oms_.cancel_order(ask_order_id_);
            ask_order_id_ = 0;
        }
    }
};
```

## 風險管理

### 風險檢查器

```cpp
class RiskManager {
    struct Limits {
        double max_position_value;
        double max_order_value;
        uint64_t max_order_quantity;
        double max_notional_per_symbol;
    };
    
    Limits limits_;
    
    struct PositionInfo {
        int64_t quantity;  // 正數=多頭, 負數=空頭
        double avg_price;
        double realized_pnl;
        double unrealized_pnl;
    };
    
    std::unordered_map<std::string, PositionInfo> positions_;
    mutable std::mutex mutex_;
    
public:
    explicit RiskManager(double max_pos_val, double max_ord_val)
        : limits_{max_pos_val, max_ord_val, 100000, max_pos_val / 10} {}
    
    bool check_order(const std::string& symbol, Side side, 
                     double price, uint64_t quantity) {
        std::lock_guard<std::mutex> lock(mutex_);
        
        // 檢查訂單價值
        double order_value = price * quantity;
        if (order_value > limits_.max_order_value) {
            return false;
        }
        
        // 檢查訂單數量
        if (quantity > limits_.max_order_quantity) {
            return false;
        }
        
        // 檢查持倉限制
        auto it = positions_.find(symbol);
        if (it != positions_.end()) {
            int64_t new_quantity = it->second.quantity;
            
            if (side == Side::BUY) {
                new_quantity += quantity;
            } else {
                new_quantity -= quantity;
            }
            
            double new_value = std::abs(new_quantity) * price;
            
            if (new_value > limits_.max_notional_per_symbol) {
                return false;
            }
        }
        
        // 檢查總持倉價值
        double total_value = calculate_total_position_value();
        if (total_value + order_value > limits_.max_position_value) {
            return false;
        }
        
        return true;
    }
    
    void update_position(const std::string& symbol, Side side,
                        double price, uint64_t quantity) {
        std::lock_guard<std::mutex> lock(mutex_);
        
        auto& pos = positions_[symbol];
        
        int64_t signed_qty = (side == Side::BUY) ? 
            static_cast<int64_t>(quantity) : -static_cast<int64_t>(quantity);
        
        if ((pos.quantity > 0 && signed_qty > 0) ||
            (pos.quantity < 0 && signed_qty < 0)) {
            // 增加持倉
            double total_cost = pos.avg_price * std::abs(pos.quantity) + 
                               price * quantity;
            pos.quantity += signed_qty;
            pos.avg_price = total_cost / std::abs(pos.quantity);
        } else {
            // 減少持倉或反向
            int64_t old_qty = pos.quantity;
            pos.quantity += signed_qty;
            
            if ((old_qty > 0 && pos.quantity < 0) ||
                (old_qty < 0 && pos.quantity > 0)) {
                // 完全反向
                pos.avg_price = price;
            }
        }
    }
    
    double get_pnl(const std::string& symbol, double current_price) const {
        std::lock_guard<std::mutex> lock(mutex_);
        
        auto it = positions_.find(symbol);
        if (it == positions_.end()) {
            return 0.0;
        }
        
        return (current_price - it->second.avg_price) * it->second.quantity;
    }
    
private:
    double calculate_total_position_value() const {
        double total = 0.0;
        
        for (const auto& [symbol, pos] : positions_) {
            total += std::abs(pos.quantity * pos.avg_price);
        }
        
        return total;
    }
};
```

## 訂單管理系統

### OMS 實現

```cpp
class OrderManagementSystem {
public:
    using OrderCallback = std::function<void(const OrderUpdate&)>;
    
private:
    struct OrderInfo {
        uint64_t order_id;
        std::string symbol;
        Side side;
        double price;
        uint64_t quantity;
        uint64_t filled_quantity;
        OrderStatus status;
        uint64_t timestamp;
    };
    
    std::unordered_map<uint64_t, OrderInfo> orders_;
    std::mutex mutex_;
    
    std::vector<OrderCallback> callbacks_;
    
    ExecutionEngine& execution_;
    RiskManager& risk_manager_;
    
    std::atomic<uint64_t> next_order_id_{1};
    
public:
    OrderManagementSystem(ExecutionEngine& exec, RiskManager& risk)
        : execution_(exec), risk_manager_(risk) {}
    
    uint64_t submit_order(const std::string& symbol, Side side,
                         double price, uint64_t quantity) {
        // 風險檢查
        if (!risk_manager_.check_order(symbol, side, price, quantity)) {
            return 0;  // 風險拒絕
        }
        
        uint64_t order_id = next_order_id_.fetch_add(1);
        
        OrderInfo info;
        info.order_id = order_id;
        info.symbol = symbol;
        info.side = side;
        info.price = price;
        info.quantity = quantity;
        info.filled_quantity = 0;
        info.status = OrderStatus::PENDING_NEW;
        info.timestamp = get_timestamp();
        
        {
            std::lock_guard<std::mutex> lock(mutex_);
            orders_[order_id] = info;
        }
        
        // 發送到執行引擎
        execution_.submit_order(order_id, symbol, side, price, quantity);
        
        return order_id;
    }
    
    bool cancel_order(uint64_t order_id) {
        std::lock_guard<std::mutex> lock(mutex_);
        
        auto it = orders_.find(order_id);
        if (it == orders_.end()) {
            return false;
        }
        
        if (it->second.status != OrderStatus::NEW &&
            it->second.status != OrderStatus::PARTIALLY_FILLED) {
            return false;  // 無法取消
        }
        
        execution_.cancel_order(order_id);
        
        return true;
    }
    
    void handle_order_update(const OrderUpdate& update) {
        {
            std::lock_guard<std::mutex> lock(mutex_);
            
            auto it = orders_.find(update.order_id);
            if (it != orders_.end()) {
                it->second.status = update.status;
                it->second.filled_quantity = update.filled_quantity;
                
                // 更新持倉
                if (update.filled_quantity > 0) {
                    risk_manager_.update_position(
                        it->second.symbol,
                        it->second.side,
                        update.fill_price,
                        update.filled_quantity
                    );
                }
            }
        }
        
        // 通知回調
        for (const auto& callback : callbacks_) {
            callback(update);
        }
    }
    
    void register_callback(OrderCallback callback) {
        callbacks_.push_back(std::move(callback));
    }
    
private:
    static uint64_t get_timestamp() {
        return std::chrono::duration_cast<std::chrono::nanoseconds>(
            std::chrono::system_clock::now().time_since_epoch()
        ).count();
    }
};
```

```cpp
class BacktestEngine {
    struct HistoricalTick {
        uint64_t timestamp;
        std::string symbol;
        double bid;
        double ask;
        uint64_t volume;
    };
    
    std::vector<HistoricalTick> historical_data_;
    TradingStrategy& strategy_;
    
    struct BacktestResults {
        double total_pnl;
        double sharpe_ratio;
        uint64_t total_trades;
        uint64_t winning_trades;
        double max_drawdown;
    };
    
public:
    BacktestEngine(TradingStrategy& strategy) : strategy_(strategy) {}
    
    void load_data(const std::string& filepath) {
        // 加載歷史數據
    }
    
    BacktestResults run() {
        strategy_.on_start();
        
        BacktestResults results{};
        
        for (const auto& tick : historical_data_) {
            MarketData md;
            md.timestamp = tick.timestamp;
            md.symbol = tick.symbol;
            md.bid_price = tick.bid;
            md.ask_price = tick.ask;
            
            strategy_.on_market_data(md);
        }
        
        strategy_.on_stop();
        
        return results;
    }
};
```

```cpp
int main() {
    // 配置
    TradingSystem::Config config;
    config.market_data_host = "239.1.1.1";
    config.market_data_port = 10000;
    config.order_gateway_host = "192.168.1.100";
    config.order_gateway_port = 8080;
    config.max_position_value = 1000000.0;
    config.max_order_value = 100000.0;
    
    // 創建系統
    TradingSystem system(config);
    
    // 創建策略
    auto strategy = std::make_shared<SimpleMarketMaker>(
        "AAPL", 0.02, 100, system.oms()
    );
    
    // 註冊策略
    system.register_strategy(strategy);
    
    // 啟動系統
    system.start();
    
    // 運行...
    std::this_thread::sleep_for(std::chrono::hours(6));
    
    // 停止系統
    system.stop();
    
    return 0;
}
```
