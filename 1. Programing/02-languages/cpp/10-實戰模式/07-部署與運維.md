# 部署與運維

## 概述

本文涵蓋交易系統的部署流程、運維監控、性能調優與問題診斷。

### 運維目標

| 指標 | 目標 | 說明 |
|-----|------|------|
| 部署時間 | < 5 分鐘 | 從代碼到生產 |
| 回滾時間 | < 2 分鐘 | 出問題快速恢復 |
| 監控延遲 | < 1 秒 | 實時指標可見 |
| 日誌保留 | 90 天 | 合規要求 |

## 構建與打包

### CMake 配置

```cmake
cmake_minimum_required(VERSION 3.15)
project(TradingSystem VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Release 配置
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -DNDEBUG -flto")

# Debug 配置
set(CMAKE_CXX_FLAGS_DEBUG "-O0 -g -fsanitize=address -fsanitize=undefined")

# 尋找依賴
find_package(spdlog REQUIRED)
find_package(nlohmann_json REQUIRED)

# 主執行檔
add_executable(trading_engine
    src/main.cpp
    src/market_data.cpp
    src/order_management.cpp
    src/execution_engine.cpp
    src/risk_manager.cpp
)

target_link_libraries(trading_engine
    spdlog::spdlog
    nlohmann_json::nlohmann_json
    pthread
)

# 安裝規則
install(TARGETS trading_engine DESTINATION bin)
install(DIRECTORY config/ DESTINATION etc/trading_system)
```

### 配置加載器

```cpp
#include <fstream>
#include <nlohmann/json.hpp>

class ConfigLoader {
public:
    static nlohmann::json load(const std::string& filepath) {
        std::ifstream file(filepath);
        if (!file.is_open()) {
            throw std::runtime_error("Failed to open config: " + filepath);
        }
        
        nlohmann::json config;
        file >> config;
        
        validate(config);
        
        return config;
    }
    
private:
    static void validate(const nlohmann::json& config) {
        // 驗證必要欄位
        if (!config.contains("system")) {
            throw std::runtime_error("Missing 'system' section");
        }
        
        if (!config.contains("market_data")) {
            throw std::runtime_error("Missing 'market_data' section");
        }
        
        // 驗證數值範圍
        if (config["risk"]["max_position_value"] <= 0) {
            throw std::runtime_error("Invalid max_position_value");
        }
    }
};
```

## 系統優化

### CPU 親和性設定

```cpp
#include <pthread.h>
#include <sched.h>

class CPUAffinity {
public:
    static bool set_affinity(const std::vector<int>& cpus) {
        cpu_set_t cpuset;
        CPU_ZERO(&cpuset);
        
        for (int cpu : cpus) {
            CPU_SET(cpu, &cpuset);
        }
        
        return pthread_setaffinity_np(pthread_self(), 
                                     sizeof(cpu_set_t), 
                                     &cpuset) == 0;
    }
    
    static bool set_realtime_priority(int priority) {
        sched_param param;
        param.sched_priority = priority;  // 1-99
        
        return pthread_setschedparam(pthread_self(), 
                                    SCHED_FIFO, 
                                    &param) == 0;
    }
};

// 啟動腳本設定
void initialize_performance() {
    nlohmann::json config = ConfigLoader::load("/etc/trading/config.json");
    
    auto cpu_list = config["performance"]["cpu_affinity"]
        .get<std::vector<int>>();
    
    if (!CPUAffinity::set_affinity(cpu_list)) {
        spdlog::warn("Failed to set CPU affinity");
    }
    
    if (!CPUAffinity::set_realtime_priority(80)) {
        spdlog::warn("Failed to set realtime priority (requires root)");
    }
}
```

```cpp
#include <sys/mman.h>

void* allocate_huge_pages(size_t size) {
    void* ptr = mmap(nullptr, size, 
                    PROT_READ | PROT_WRITE,
                    MAP_PRIVATE | MAP_ANONYMOUS | MAP_HUGETLB,
                    -1, 0);
    
    if (ptr == MAP_FAILED) {
        spdlog::error("Failed to allocate huge pages");
        return nullptr;
    }
    
    return ptr;
}
```

### 日誌配置

```cpp
#include <spdlog/spdlog.h>
#include <spdlog/sinks/rotating_file_sink.h>
#include <spdlog/sinks/syslog_sink.h>

void setup_logging(const nlohmann::json& config) {
    std::string log_dir = config["system"]["log_directory"];
    std::string log_level = config["system"]["log_level"];
    
    // 交易日誌 (50MB, 10 個檔案)
    auto trade_logger = spdlog::rotating_logger_mt(
        "trade",
        log_dir + "/trades.log",
        1024 * 1024 * 50,
        10
    );
    
    // 系統日誌
    auto sys_logger = spdlog::rotating_logger_mt(
        "system",
        log_dir + "/system.log",
        1024 * 1024 * 10,
        5
    );
    
    // 錯誤日誌 (同時寫入檔案和 syslog)
    std::vector<spdlog::sink_ptr> sinks;
    sinks.push_back(std::make_shared<spdlog::sinks::rotating_file_sink_mt>(
        log_dir + "/errors.log", 1024 * 1024 * 10, 3));
    sinks.push_back(std::make_shared<spdlog::sinks::syslog_sink_mt>(
        "trading_system", LOG_PID, LOG_USER, false));
    
    auto error_logger = std::make_shared<spdlog::logger>(
        "error", sinks.begin(), sinks.end());
    
    spdlog::register_logger(error_logger);
    
    // 設定日誌等級
    if (log_level == "debug") {
        spdlog::set_level(spdlog::level::debug);
    } else if (log_level == "info") {
        spdlog::set_level(spdlog::level::info);
    }
}
```

## 監控系統

### Prometheus 指標導出

```cpp
#include <prometheus/exposer.h>
#include <prometheus/registry.h>
#include <prometheus/counter.h>
#include <prometheus/gauge.h>
#include <prometheus/histogram.h>

class MetricsExporter {
    std::shared_ptr<prometheus::Registry> registry_;
    prometheus::Exposer exposer_;
    
    // 計數器
    prometheus::Family<prometheus::Counter>& orders_counter_;
    prometheus::Counter& orders_submitted_;
    prometheus::Counter& orders_filled_;
    
    // 直方圖
    prometheus::Family<prometheus::Histogram>& latency_histogram_;
    prometheus::Histogram& order_latency_;
    
public:
    MetricsExporter() 
        : registry_(std::make_shared<prometheus::Registry>()),
          exposer_("0.0.0.0:9090"),
          orders_counter_(prometheus::BuildCounter()
              .Name("orders_total")
              .Help("Total number of orders")
              .Register(*registry_)),
          orders_submitted_(orders_counter_.Add({{"status", "submitted"}})),
          orders_filled_(orders_counter_.Add({{"status", "filled"}})),
          latency_histogram_(prometheus::BuildHistogram()
              .Name("order_latency_microseconds")
              .Help("Order processing latency")
              .Register(*registry_)),
          order_latency_(latency_histogram_.Add({},
              prometheus::Histogram::BucketBoundaries{10, 50, 100, 500, 1000, 5000})) {
        
        exposer_.RegisterCollectable(registry_);
    }
    
    void record_order_submitted() {
        orders_submitted_.Increment();
    }
    
    void record_order_filled() {
        orders_filled_.Increment();
    }
    
    void record_latency(double latency_us) {
        order_latency_.Observe(latency_us);
    }
};
```

### 部署腳本

```bash
#!/bin/bash
# deploy.sh

set -e

VERSION=$1
if [ -z "$VERSION" ]; then
    echo "Usage: $0 <version>"
    exit 1
fi

DEPLOY_DIR="/opt/trading"
BACKUP_DIR="/opt/trading_backups"

echo "Deploying version ${VERSION}..."

# 1. 備份當前版本
echo "Creating backup..."
mkdir -p ${BACKUP_DIR}
tar -czf ${BACKUP_DIR}/backup_$(date +%Y%m%d_%H%M%S).tar.gz \
    -C ${DEPLOY_DIR} .

# 2. 停止服務
echo "Stopping service..."
sudo systemctl stop trading-system

# 3. 部署新版本
echo "Deploying new version..."
tar -xzf trading_system_${VERSION}.tar.gz -C ${DEPLOY_DIR}

# 4. 運行健康檢查
echo "Running health check..."
${DEPLOY_DIR}/bin/trading_engine --check-config /etc/trading/config.json

# 5. 啟動服務
echo "Starting service..."
sudo systemctl start trading-system

# 6. 驗證
sleep 5
if sudo systemctl is-active --quiet trading-system; then
    echo "Deployment successful!"
else
    echo "Deployment failed, rolling back..."
    sudo systemctl stop trading-system
    
    # 恢復備份
    LATEST_BACKUP=$(ls -t ${BACKUP_DIR}/backup_*.tar.gz | head -1)
    tar -xzf ${LATEST_BACKUP} -C ${DEPLOY_DIR}
    
    sudo systemctl start trading-system
    exit 1
fi

# 7. 清理舊備份 (保留最近 10 個)
ls -t ${BACKUP_DIR}/backup_*.tar.gz | tail -n +11 | xargs -r rm
```

## 健康檢查

### 健康檢查端點

```cpp
#include <httplib.h>

class HealthCheckServer {
    httplib::Server server_;
    std::atomic<bool> is_healthy_{true};
    
public:
    void start(int port = 8888) {
        server_.Get("/health", [this](const httplib::Request&, httplib::Response& res) {
            if (is_healthy_.load()) {
                res.set_content(R"({"status": "healthy"})", "application/json");
                res.status = 200;
            } else {
                res.set_content(R"({"status": "unhealthy"})", "application/json");
                res.status = 503;
            }
        });
        
        server_.Get("/metrics", [](const httplib::Request&, httplib::Response& res) {
            // 返回 Prometheus 格式指標
            res.set_content(collect_metrics(), "text/plain");
        });
        
        server_.listen("0.0.0.0", port);
    }
    
    void set_healthy(bool healthy) {
        is_healthy_.store(healthy);
    }
    
private:
    static std::string collect_metrics() {
        // 收集指標
        return "# HELP orders_total Total orders\n"
               "# TYPE orders_total counter\n"
               "orders_total 12345\n";
    }
};
```

## 問題診斷

### 性能分析

```bash
#!/bin/bash
# analyze_performance.sh

PID=$(pgrep trading_engine)

if [ -z "$PID" ]; then
    echo "Trading engine not running"
    exit 1
fi

echo "Analyzing PID ${PID}..."

# CPU 使用率
echo "=== CPU Usage ==="
top -b -n 1 -p ${PID} | tail -2

# 記憶體使用
echo "=== Memory Usage ==="
cat /proc/${PID}/status | grep -E "VmSize|VmRSS|VmData"

# 執行緒數
echo "=== Thread Count ==="
ls /proc/${PID}/task | wc -l

# 網路連接
echo "=== Network Connections ==="
ss -tunap | grep ${PID}

# 檔案描述符
echo "=== Open File Descriptors ==="
ls /proc/${PID}/fd | wc -l
lsof -p ${PID} | head -20

# perf 採樣 (5 秒)
echo "=== CPU Profile (5s) ==="
sudo perf record -p ${PID} -g sleep 5
sudo perf report --stdio | head -50
```

```cpp
#include <fstream>

class LatencyTracer {
    std::ofstream trace_file_;
    
public:
    struct Trace {
        const char* component;
        uint64_t start_tsc;
        
        Trace(const char* comp) : component(comp) {
            start_tsc = __rdtsc();
        }
        
        ~Trace() {
            uint64_t end_tsc = __rdtsc();
            uint64_t cycles = end_tsc - start_tsc;
            
            // 記錄 (component, cycles)
            // 實際應該使用無鎖環形緩衝區
        }
    };
    
    static Trace trace(const char* component) {
        return Trace(component);
    }
};

// 使用範例
```

### 演練腳本

```bash
#!/bin/bash
# disaster_drill.sh

echo "=== Disaster Recovery Drill ==="

# 場景 1: 主節點故障
echo "Scenario 1: Primary node failure"
sudo systemctl stop trading-system@primary
sleep 5

# 檢查備用節點是否接管
if systemctl is-active --quiet trading-system@backup; then
    echo "✓ Backup node active"
else
    echo "✗ Backup node failed to activate"
fi

# 恢復主節點
sudo systemctl start trading-system@primary

# 場景 2: 網路分區
echo "Scenario 2: Network partition"
sudo iptables -A INPUT -s 192.168.1.100 -j DROP
sleep 10
sudo iptables -D INPUT -s 192.168.1.100 -j DROP

# 場景 3: 記憶體耗盡模擬
echo "Scenario 3: Memory pressure"
stress-ng --vm 1 --vm-bytes 80% --timeout 30s

echo "Drill completed"
```
