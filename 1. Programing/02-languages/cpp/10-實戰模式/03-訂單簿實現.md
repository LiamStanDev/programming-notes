# 訂單簿實現

## 概述

訂單簿 (Order Book) 是交易系統的核心數據結構，維護所有買賣訂單的即時狀態。本文涵蓋高效能訂單簿的設計與實現。

### 核心需求

| 需求 | 說明 | 典型值 |
|-----|------|-------|
| 添加訂單 | 新訂單插入 | < 100 ns |
| 刪除訂單 | 取消訂單 | < 100 ns |
| 修改訂單 | 價格/數量更新 | < 100 ns |
| 最佳報價 | Best Bid/Ask | < 10 ns |
| 深度查詢 | N 檔報價 | < 500 ns |

### 數據結構選擇

```
訂單簿結構:
┌─────────────────────────────────────┐
│          Order Book                 │
├──────────────┬──────────────────────┤
│  Bid Side    │     Ask Side         │
├──────────────┼──────────────────────┤
│ Price Levels │   Price Levels       │
│  (降序)      │    (升序)            │
│              │                      │
│ 150.50: 1000 │   150.51: 800       │
│ 150.49: 1500 │   150.52: 1200      │
│ 150.48: 2000 │   150.53: 900       │
└──────────────┴──────────────────────┘
```

## 基礎訂單簿

### 簡單實現

```cpp
#include <map>
#include <unordered_map>
#include <cstdint>

struct Order {
    uint64_t order_id;
    uint32_t symbol_id;
    double price;
    uint64_t quantity;
    char side;  // 'B' or 'S'
    
    Order(uint64_t id, uint32_t sym, double p, uint64_t q, char s)
        : order_id(id), symbol_id(sym), price(p), quantity(q), side(s) {}
};

class SimpleOrderBook {
    // 價格 -> 總數量
    std::map<double, uint64_t, std::greater<>> bids_;  // 降序
    std::map<double, uint64_t> asks_;                   // 升序
    
    // 訂單 ID -> 訂單
    std::unordered_map<uint64_t, Order> orders_;
    
public:
    void add_order(uint64_t order_id, uint32_t symbol_id, 
                   double price, uint64_t quantity, char side) {
        
        orders_.emplace(order_id, Order(order_id, symbol_id, price, quantity, side));
        
        if (side == 'B') {
            bids_[price] += quantity;
        } else {
            asks_[price] += quantity;
        }
    }
    
    bool cancel_order(uint64_t order_id) {
        auto it = orders_.find(order_id);
        if (it == orders_.end()) {
            return false;
        }
        
        const Order& order = it->second;
        
        if (order.side == 'B') {
            bids_[order.price] -= order.quantity;
            if (bids_[order.price] == 0) {
                bids_.erase(order.price);
            }
        } else {
            asks_[order.price] -= order.quantity;
            if (asks_[order.price] == 0) {
                asks_.erase(order.price);
            }
        }
        
        orders_.erase(it);
        return true;
    }
    
    bool modify_order(uint64_t order_id, uint64_t new_quantity) {
        auto it = orders_.find(order_id);
        if (it == orders_.end()) {
            return false;
        }
        
        Order& order = it->second;
        
        if (order.side == 'B') {
            bids_[order.price] = bids_[order.price] - order.quantity + new_quantity;
        } else {
            asks_[order.price] = asks_[order.price] - order.quantity + new_quantity;
        }
        
        order.quantity = new_quantity;
        return true;
    }
    
    double best_bid() const {
        return bids_.empty() ? 0.0 : bids_.begin()->first;
    }
    
    double best_ask() const {
        return asks_.empty() ? 0.0 : asks_.begin()->first;
    }
    
    double spread() const {
        if (bids_.empty() || asks_.empty()) {
            return 0.0;
        }
        return best_ask() - best_bid();
    }
    
    uint64_t bid_volume(double price) const {
        auto it = bids_.find(price);
        return (it != bids_.end()) ? it->second : 0;
    }
    
    uint64_t ask_volume(double price) const {
        auto it = asks_.find(price);
        return (it != asks_.end()) ? it->second : 0;
    }
};
```

## 高效能訂單簿

### 價格檔位結構

```cpp
#include <list>
#include <unordered_map>

struct PriceLevel {
    double price;
    uint64_t total_quantity;
    std::list<uint64_t> order_queue;  // FIFO 訂單隊列
    
    explicit PriceLevel(double p) : price(p), total_quantity(0) {}
};

class OrderBookV2 {
    struct OrderInfo {
        double price;
        uint64_t quantity;
        char side;
        std::list<uint64_t>::iterator queue_position;
        PriceLevel* level;
    };
    
    // 價格 -> 檔位
    std::map<double, PriceLevel, std::greater<>> bid_levels_;
    std::map<double, PriceLevel> ask_levels_;
    
    // 訂單 ID -> 訂單信息
    std::unordered_map<uint64_t, OrderInfo> orders_;
    
    // 快取最佳報價
    PriceLevel* best_bid_level_{nullptr};
    PriceLevel* best_ask_level_{nullptr};
    
public:
    void add_order(uint64_t order_id, double price, uint64_t quantity, char side) {
        if (side == 'B') {
            auto [it, inserted] = bid_levels_.try_emplace(price, price);
            PriceLevel& level = it->second;
            
            level.order_queue.push_back(order_id);
            level.total_quantity += quantity;
            
            OrderInfo info;
            info.price = price;
            info.quantity = quantity;
            info.side = 'B';
            info.queue_position = --level.order_queue.end();
            info.level = &level;
            
            orders_[order_id] = info;
            
            update_best_bid();
            
        } else {
            auto [it, inserted] = ask_levels_.try_emplace(price, price);
            PriceLevel& level = it->second;
            
            level.order_queue.push_back(order_id);
            level.total_quantity += quantity;
            
            OrderInfo info;
            info.price = price;
            info.quantity = quantity;
            info.side = 'S';
            info.queue_position = --level.order_queue.end();
            info.level = &level;
            
            orders_[order_id] = info;
            
            update_best_ask();
        }
    }
    
    bool cancel_order(uint64_t order_id) {
        auto it = orders_.find(order_id);
        if (it == orders_.end()) {
            return false;
        }
        
        OrderInfo& info = it->second;
        PriceLevel* level = info.level;
        
        level->total_quantity -= info.quantity;
        level->order_queue.erase(info.queue_position);
        
        // 如果檔位空了，移除
        if (level->order_queue.empty()) {
            if (info.side == 'B') {
                bid_levels_.erase(info.price);
                update_best_bid();
            } else {
                ask_levels_.erase(info.price);
                update_best_ask();
            }
        }
        
        orders_.erase(it);
        return true;
    }
    
    bool modify_quantity(uint64_t order_id, uint64_t new_quantity) {
        auto it = orders_.find(order_id);
        if (it == orders_.end()) {
            return false;
        }
        
        OrderInfo& info = it->second;
        PriceLevel* level = info.level;
        
        level->total_quantity = level->total_quantity - info.quantity + new_quantity;
        info.quantity = new_quantity;
        
        return true;
    }
    
    double best_bid() const {
        return best_bid_level_ ? best_bid_level_->price : 0.0;
    }
    
    double best_ask() const {
        return best_ask_level_ ? best_ask_level_->price : 0.0;
    }
    
    uint64_t best_bid_quantity() const {
        return best_bid_level_ ? best_bid_level_->total_quantity : 0;
    }
    
    uint64_t best_ask_quantity() const {
        return best_ask_level_ ? best_ask_level_->total_quantity : 0;
    }
    
private:
    void update_best_bid() {
        best_bid_level_ = bid_levels_.empty() ? nullptr : &bid_levels_.begin()->second;
    }
    
    void update_best_ask() {
        best_ask_level_ = ask_levels_.empty() ? nullptr : &ask_levels_.begin()->second;
    }
};
```

## 固定價格網格訂單簿

### Tick-Based 實現

```cpp
#include <vector>
#include <array>
#include <cstdint>

template<size_t MaxTicks = 100000>
class TickBasedOrderBook {
    struct Level {
        uint64_t total_quantity{0};
        uint32_t order_count{0};
    };
    
    // 價格檔位數組
    std::array<Level, MaxTicks> bid_levels_;
    std::array<Level, MaxTicks> ask_levels_;
    
    // 訂單信息
    struct OrderInfo {
        uint32_t tick_index;
        uint64_t quantity;
        char side;
    };
    
    std::unordered_map<uint64_t, OrderInfo> orders_;
    
    // 最佳報價索引
    uint32_t best_bid_tick_{0};
    uint32_t best_ask_tick_{MaxTicks - 1};
    
    // 價格轉換
    double tick_size_;
    double base_price_;
    
public:
    TickBasedOrderBook(double tick_size, double base_price)
        : tick_size_(tick_size), base_price_(base_price) {}
    
    uint32_t price_to_tick(double price) const {
        return static_cast<uint32_t>((price - base_price_) / tick_size_);
    }
    
    double tick_to_price(uint32_t tick) const {
        return base_price_ + tick * tick_size_;
    }
    
    void add_order(uint64_t order_id, double price, uint64_t quantity, char side) {
        uint32_t tick = price_to_tick(price);
        
        if (tick >= MaxTicks) {
            return;  // 超出範圍
        }
        
        if (side == 'B') {
            bid_levels_[tick].total_quantity += quantity;
            bid_levels_[tick].order_count++;
            
            if (tick > best_bid_tick_) {
                best_bid_tick_ = tick;
            }
        } else {
            ask_levels_[tick].total_quantity += quantity;
            ask_levels_[tick].order_count++;
            
            if (tick < best_ask_tick_) {
                best_ask_tick_ = tick;
            }
        }
        
        orders_[order_id] = OrderInfo{tick, quantity, side};
    }
    
    bool cancel_order(uint64_t order_id) {
        auto it = orders_.find(order_id);
        if (it == orders_.end()) {
            return false;
        }
        
        const OrderInfo& info = it->second;
        
        if (info.side == 'B') {
            Level& level = bid_levels_[info.tick_index];
            level.total_quantity -= info.quantity;
            level.order_count--;
            
            if (level.order_count == 0 && info.tick_index == best_bid_tick_) {
                update_best_bid();
            }
        } else {
            Level& level = ask_levels_[info.tick_index];
            level.total_quantity -= info.quantity;
            level.order_count--;
            
            if (level.order_count == 0 && info.tick_index == best_ask_tick_) {
                update_best_ask();
            }
        }
        
        orders_.erase(it);
        return true;
    }
    
    double best_bid() const {
        return tick_to_price(best_bid_tick_);
    }
    
    double best_ask() const {
        return tick_to_price(best_ask_tick_);
    }
    
    uint64_t bid_quantity_at_tick(uint32_t tick) const {
        return tick < MaxTicks ? bid_levels_[tick].total_quantity : 0;
    }
    
    uint64_t ask_quantity_at_tick(uint32_t tick) const {
        return tick < MaxTicks ? ask_levels_[tick].total_quantity : 0;
    }
    
    // 獲取市場深度 (前 N 檔)
    std::vector<std::pair<double, uint64_t>> get_bid_depth(size_t levels) const {
        std::vector<std::pair<double, uint64_t>> result;
        result.reserve(levels);
        
        uint32_t tick = best_bid_tick_;
        
        while (result.size() < levels && tick > 0) {
            if (bid_levels_[tick].order_count > 0) {
                result.emplace_back(
                    tick_to_price(tick),
                    bid_levels_[tick].total_quantity
                );
            }
            --tick;
        }
        
        return result;
    }
    
    std::vector<std::pair<double, uint64_t>> get_ask_depth(size_t levels) const {
        std::vector<std::pair<double, uint64_t>> result;
        result.reserve(levels);
        
        uint32_t tick = best_ask_tick_;
        
        while (result.size() < levels && tick < MaxTicks) {
            if (ask_levels_[tick].order_count > 0) {
                result.emplace_back(
                    tick_to_price(tick),
                    ask_levels_[tick].total_quantity
                );
            }
            ++tick;
        }
        
        return result;
    }
    
private:
    void update_best_bid() {
        while (best_bid_tick_ > 0 && bid_levels_[best_bid_tick_].order_count == 0) {
            --best_bid_tick_;
        }
    }
    
    void update_best_ask() {
        while (best_ask_tick_ < MaxTicks - 1 && 
               ask_levels_[best_ask_tick_].order_count == 0) {
            ++best_ask_tick_;
        }
    }
};
```

## 撮合引擎

### 簡單撮合實現

```cpp
#include <vector>

struct Trade {
    uint64_t buy_order_id;
    uint64_t sell_order_id;
    double price;
    uint64_t quantity;
    uint64_t timestamp;
};

class MatchingEngine {
    OrderBookV2 order_book_;
    std::vector<Trade> trades_;
    
public:
    struct MatchResult {
        bool fully_matched;
        uint64_t matched_quantity;
        std::vector<Trade> trades;
    };
    
    MatchResult submit_market_order(uint64_t order_id, uint64_t quantity, char side) {
        MatchResult result;
        result.fully_matched = false;
        result.matched_quantity = 0;
        
        uint64_t remaining = quantity;
        
        if (side == 'B') {
            // 買單: 與最佳賣價撮合
            while (remaining > 0) {
                double best_ask = order_book_.best_ask();
                if (best_ask == 0.0) {
                    break;  // 沒有賣單
                }
                
                uint64_t available = order_book_.best_ask_quantity();
                uint64_t matched = std::min(remaining, available);
                
                // 創建成交
                Trade trade;
                trade.buy_order_id = order_id;
                trade.sell_order_id = 0;  // 簡化: 未追蹤對手訂單
                trade.price = best_ask;
                trade.quantity = matched;
                trade.timestamp = get_timestamp();
                
                result.trades.push_back(trade);
                result.matched_quantity += matched;
                remaining -= matched;
                
                // 更新訂單簿
                // (實際需要移除/修改被撮合的訂單)
            }
            
        } else {
            // 賣單: 與最佳買價撮合
            while (remaining > 0) {
                double best_bid = order_book_.best_bid();
                if (best_bid == 0.0) {
                    break;
                }
                
                uint64_t available = order_book_.best_bid_quantity();
                uint64_t matched = std::min(remaining, available);
                
                Trade trade;
                trade.buy_order_id = 0;
                trade.sell_order_id = order_id;
                trade.price = best_bid;
                trade.quantity = matched;
                trade.timestamp = get_timestamp();
                
                result.trades.push_back(trade);
                result.matched_quantity += matched;
                remaining -= matched;
            }
        }
        
        result.fully_matched = (remaining == 0);
        
        return result;
    }
    
    void submit_limit_order(uint64_t order_id, double price, 
                           uint64_t quantity, char side) {
        
        // 嘗試立即撮合
        if (side == 'B') {
            double best_ask = order_book_.best_ask();
            
            if (best_ask > 0.0 && price >= best_ask) {
                // 可以撮合
                auto result = submit_market_order(order_id, quantity, side);
                
                uint64_t remaining = quantity - result.matched_quantity;
                
                if (remaining > 0) {
                    // 剩餘部分掛單
                    order_book_.add_order(order_id, price, remaining, side);
                }
                
                return;
            }
        } else {
            double best_bid = order_book_.best_bid();
            
            if (best_bid > 0.0 && price <= best_bid) {
                auto result = submit_market_order(order_id, quantity, side);
                
                uint64_t remaining = quantity - result.matched_quantity;
                
                if (remaining > 0) {
                    order_book_.add_order(order_id, price, remaining, side);
                }
                
                return;
            }
        }
        
        // 無法撮合，直接掛單
        order_book_.add_order(order_id, price, quantity, side);
    }
    
private:
    static uint64_t get_timestamp() {
        return std::chrono::duration_cast<std::chrono::nanoseconds>(
            std::chrono::system_clock::now().time_since_epoch()
        ).count();
    }
};
```

## 訂單簿快照與增量更新

### 快照管理

```cpp
#include <vector>

struct OrderBookSnapshot {
    uint64_t sequence_number;
    uint64_t timestamp;
    
    struct Level {
        double price;
        uint64_t quantity;
    };
    
    std::vector<Level> bids;
    std::vector<Level> asks;
};

class SnapshotManager {
    OrderBookV2& order_book_;
    uint64_t snapshot_sequence_{0};
    
public:
    explicit SnapshotManager(OrderBookV2& book) : order_book_(book) {}
    
    OrderBookSnapshot take_snapshot(size_t depth = 10) {
        OrderBookSnapshot snapshot;
        snapshot.sequence_number = ++snapshot_sequence_;
        snapshot.timestamp = get_timestamp();
        
        // 獲取買賣深度
        auto bid_depth = order_book_.get_bid_depth(depth);
        auto ask_depth = order_book_.get_ask_depth(depth);
        
        snapshot.bids.reserve(bid_depth.size());
        for (const auto& [price, qty] : bid_depth) {
            snapshot.bids.push_back({price, qty});
        }
        
        snapshot.asks.reserve(ask_depth.size());
        for (const auto& [price, qty] : ask_depth) {
            snapshot.asks.push_back({price, qty});
        }
        
        return snapshot;
    }
    
    static void print_snapshot(const OrderBookSnapshot& snapshot) {
        std::cout << "Snapshot #" << snapshot.sequence_number << "\n";
        std::cout << "Timestamp: " << snapshot.timestamp << "\n\n";
        
        std::cout << "Asks:\n";
        for (auto it = snapshot.asks.rbegin(); it != snapshot.asks.rend(); ++it) {
            std::cout << "  " << it->price << " : " << it->quantity << "\n";
        }
        
        std::cout << "---\n";
        
        std::cout << "Bids:\n";
        for (const auto& level : snapshot.bids) {
            std::cout << "  " << level.price << " : " << level.quantity << "\n";
        }
        
        std::cout << "\n";
    }
    
private:
    static uint64_t get_timestamp() {
        return std::chrono::duration_cast<std::chrono::microseconds>(
            std::chrono::system_clock::now().time_since_epoch()
        ).count();
    }
};
```

### 增量更新

```cpp
struct OrderBookUpdate {
    enum class Type {
        ADD,
        MODIFY,
        DELETE
    };
    
    Type type;
    uint64_t order_id;
    double price;
    uint64_t quantity;
    char side;
    uint64_t sequence_number;
};

class IncrementalUpdater {
    OrderBookV2& order_book_;
    uint64_t last_sequence_{0};
    
public:
    explicit IncrementalUpdater(OrderBookV2& book) : order_book_(book) {}
    
    bool apply_update(const OrderBookUpdate& update) {
        // 檢查序列號
        if (update.sequence_number != last_sequence_ + 1) {
            // 序列號不連續，需要重新同步
            return false;
        }
        
        switch (update.type) {
            case OrderBookUpdate::Type::ADD:
                order_book_.add_order(update.order_id, update.price, 
                                     update.quantity, update.side);
                break;
                
            case OrderBookUpdate::Type::MODIFY:
                order_book_.modify_quantity(update.order_id, update.quantity);
                break;
                
            case OrderBookUpdate::Type::DELETE:
                order_book_.cancel_order(update.order_id);
                break;
        }
        
        last_sequence_ = update.sequence_number;
        return true;
    }
    
    uint64_t last_sequence() const {
        return last_sequence_;
    }
};
```

## 訂單簿分析

### 成交量分析

```cpp
class VolumeAnalyzer {
    struct VolumeProfile {
        double price;
        uint64_t bid_volume;
        uint64_t ask_volume;
        uint64_t imbalance;  // bid - ask
    };
    
public:
    static std::vector<VolumeProfile> analyze(const OrderBookV2& book, size_t depth) {
        std::vector<VolumeProfile> profile;
        profile.reserve(depth * 2);
        
        auto bids = book.get_bid_depth(depth);
        auto asks = book.get_ask_depth(depth);
        
        // 合併買賣深度
        std::unordered_map<double, VolumeProfile> levels;
        
        for (const auto& [price, qty] : bids) {
            auto& p = levels[price];
            p.price = price;
            p.bid_volume = qty;
        }
        
        for (const auto& [price, qty] : asks) {
            auto& p = levels[price];
            p.price = price;
            p.ask_volume = qty;
        }
        
        // 計算不平衡度
        for (auto& [price, p] : levels) {
            p.imbalance = static_cast<int64_t>(p.bid_volume) - 
                         static_cast<int64_t>(p.ask_volume);
            profile.push_back(p);
        }
        
        // 按價格排序
        std::sort(profile.begin(), profile.end(),
            [](const VolumeProfile& a, const VolumeProfile& b) {
                return a.price < b.price;
            });
        
        return profile;
    }
    
    static double calculate_vwap(const std::vector<VolumeProfile>& profile) {
        double total_value = 0.0;
        uint64_t total_volume = 0;
        
        for (const auto& p : profile) {
            uint64_t volume = p.bid_volume + p.ask_volume;
            total_value += p.price * volume;
            total_volume += volume;
        }
        
        return total_volume > 0 ? total_value / total_volume : 0.0;
    }
};
```

### 流動性指標

```cpp
class LiquidityMetrics {
public:
    struct Metrics {
        double spread;
        double spread_bps;  // 基點
        uint64_t bid_depth_5;
        uint64_t ask_depth_5;
        double mid_price;
        double liquidity_score;
    };
    
    static Metrics calculate(const OrderBookV2& book) {
        Metrics m;
        
        double bid = book.best_bid();
        double ask = book.best_ask();
        
        m.spread = ask - bid;
        m.mid_price = (bid + ask) / 2.0;
        m.spread_bps = (m.spread / m.mid_price) * 10000.0;
        
        // 前 5 檔深度
        auto bid_depth = book.get_bid_depth(5);
        auto ask_depth = book.get_ask_depth(5);
        
        m.bid_depth_5 = 0;
        for (const auto& [price, qty] : bid_depth) {
            m.bid_depth_5 += qty;
        }
        
        m.ask_depth_5 = 0;
        for (const auto& [price, qty] : ask_depth) {
            m.ask_depth_5 += qty;
        }
        
        // 流動性得分 (簡化)
        uint64_t total_depth = m.bid_depth_5 + m.ask_depth_5;
        m.liquidity_score = total_depth / (m.spread_bps + 1.0);
        
        return m;
    }
};
```

### 訂單簿基準

```cpp
#include <chrono>
#include <random>

public:

        OrderBookV2 book;
        std::mt19937 rng(42);
        std::uniform_real_distribution<> price_dist(100.0, 200.0);
        std::uniform_int_distribution<> qty_dist(100, 10000);
        
        std::vector<uint64_t> order_ids;
        order_ids.reserve(num_orders);
        
        // 添加訂單
        auto add_start = std::chrono::steady_clock::now();
        
        for (size_t i = 0; i < num_orders; ++i) {
            double price = price_dist(rng);
            uint64_t qty = qty_dist(rng);
            char side = (i % 2 == 0) ? 'B' : 'S';
            
            book.add_order(i, price, qty, side);
            order_ids.push_back(i);
        }
        
        auto add_end = std::chrono::steady_clock::now();
        auto add_duration = std::chrono::duration_cast<std::chrono::nanoseconds>(
            add_end - add_start);
        
        std::cout << "Add " << num_orders << " orders: "
                  << add_duration.count() / num_orders << " ns/op\n";
        
        // 查詢最佳報價
        auto query_start = std::chrono::steady_clock::now();
        
        for (size_t i = 0; i < 1000000; ++i) {
            volatile double bid = book.best_bid();
            volatile double ask = book.best_ask();
        }
        
        auto query_end = std::chrono::steady_clock::now();
        auto query_duration = std::chrono::duration_cast<std::chrono::nanoseconds>(
            query_end - query_start);
        
        std::cout << "Best quote query: " << query_duration.count() / 1000000 << " ns/op\n";
        
        // 取消訂單
        auto cancel_start = std::chrono::steady_clock::now();
        
        for (uint64_t order_id : order_ids) {
            book.cancel_order(order_id);
        }
        
        auto cancel_end = std::chrono::steady_clock::now();
        auto cancel_duration = std::chrono::duration_cast<std::chrono::nanoseconds>(
            cancel_end - cancel_start);
        
        std::cout << "Cancel " << num_orders << " orders: "
                  << cancel_duration.count() / num_orders << " ns/op\n";
    }
```

## 完整範例: 多商品訂單簿

```cpp
class MultiSymbolOrderBook {
    std::unordered_map<uint32_t, std::unique_ptr<OrderBookV2>> books_;
    
public:
    OrderBookV2* get_or_create_book(uint32_t symbol_id) {
        auto it = books_.find(symbol_id);
        
        if (it != books_.end()) {
            return it->second.get();
        }
        
        auto book = std::make_unique<OrderBookV2>();
        auto* ptr = book.get();
        books_[symbol_id] = std::move(book);
        
        return ptr;
    }
    
    void add_order(uint32_t symbol_id, uint64_t order_id, 
                   double price, uint64_t quantity, char side) {
        auto* book = get_or_create_book(symbol_id);
        book->add_order(order_id, price, quantity, side);
    }
    
    bool cancel_order(uint32_t symbol_id, uint64_t order_id) {
        auto it = books_.find(symbol_id);
        if (it == books_.end()) {
            return false;
        }
        
        return it->second->cancel_order(order_id);
    }
    
    std::pair<double, double> get_best_quotes(uint32_t symbol_id) const {
        auto it = books_.find(symbol_id);
        if (it == books_.end()) {
            return {0.0, 0.0};
        }
        
        return {it->second->best_bid(), it->second->best_ask()};
    }
    
    size_t symbol_count() const {
        return books_.size();
    }
};
```
