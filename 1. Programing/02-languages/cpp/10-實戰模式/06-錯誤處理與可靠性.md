# 錯誤處理與可靠性

## 概述

交易系統的可靠性至關重要。本文涵蓋錯誤處理策略、容錯設計、狀態恢復與監控告警。

### 可靠性目標

| 指標 | 目標 | 說明 |
|-----|------|------|
| 可用性 | 99.99% | 年停機時間 < 53 分鐘 |
| MTBF | > 1000 小時 | 平均故障間隔 |
| MTTR | < 5 分鐘 | 平均恢復時間 |
| 數據完整性 | 100% | 零遺失、零重複 |

## 錯誤分類與處理

### 錯誤類型

```cpp
enum class ErrorSeverity {
    INFO,       // 資訊性
    WARNING,    // 警告，系統可繼續
    ERROR,      // 錯誤，影響功能
    CRITICAL,   // 嚴重，需要立即處理
    FATAL       // 致命，系統無法繼續
};

class TradingError {
public:
    ErrorSeverity severity;
    std::string component;
    std::string message;
    std::string stack_trace;
    uint64_t timestamp;
    std::unordered_map<std::string, std::string> context;
    
    TradingError(ErrorSeverity sev, std::string comp, std::string msg)
        : severity(sev), component(std::move(comp)), message(std::move(msg)) {
        timestamp = get_timestamp();
    }
    
    void add_context(const std::string& key, const std::string& value) {
        context[key] = value;
    }
    
private:
    static uint64_t get_timestamp() {
        return std::chrono::duration_cast<std::chrono::nanoseconds>(
            std::chrono::system_clock::now().time_since_epoch()
        ).count();
    }
};
```

### 錯誤處理器

```cpp
class ErrorHandler {
public:
    using ErrorCallback = std::function<void(const TradingError&)>;
    
private:
    std::vector<ErrorCallback> callbacks_;
    std::mutex mutex_;
    
    // 錯誤統計
    std::unordered_map<std::string, uint64_t> error_counts_;
    
public:
    void register_callback(ErrorCallback callback) {
        std::lock_guard<std::mutex> lock(mutex_);
        callbacks_.push_back(std::move(callback));
    }
    
    void handle_error(const TradingError& error) {
        {
            std::lock_guard<std::mutex> lock(mutex_);
            error_counts_[error.component]++;
        }
        
        // 記錄錯誤
        log_error(error);
        
        // 根據嚴重性採取行動
        switch (error.severity) {
            case ErrorSeverity::INFO:
            case ErrorSeverity::WARNING:
                // 僅記錄
                break;
                
            case ErrorSeverity::ERROR:
                // 通知監控系統
                notify_monitoring(error);
                break;
                
            case ErrorSeverity::CRITICAL:
                // 觸發告警
                trigger_alert(error);
                // 嘗試自動恢復
                attempt_recovery(error);
                break;
                
            case ErrorSeverity::FATAL:
                // 執行緊急停機程序
                emergency_shutdown(error);
                break;
        }
        
        // 執行回調
        for (const auto& callback : callbacks_) {
            callback(error);
        }
    }
    
private:
    void log_error(const TradingError& error) {
        // 使用 spdlog 記錄
        spdlog::error("[{}] {}: {}", 
            error.component, 
            to_string(error.severity),
            error.message
        );
    }
    
    void notify_monitoring(const TradingError& error) {
        // 發送到監控系統
    }
    
    void trigger_alert(const TradingError& error) {
        // 發送告警 (email, SMS, PagerDuty 等)
    }
    
    void attempt_recovery(const TradingError& error) {
        // 自動恢復邏輯
    }
    
    void emergency_shutdown(const TradingError& error) {
        // 緊急停機
        spdlog::critical("FATAL ERROR: {}", error.message);
        std::abort();
    }
};
```

#include <fstream>
#include <nlohmann/json.hpp>

class CheckpointManager {
    std::string checkpoint_dir_;
    
public:
    explicit CheckpointManager(std::string dir) 
        : checkpoint_dir_(std::move(dir)) {}
    
    template<typename T>
    bool save_checkpoint(const std::string& name, const T& state) {
        nlohmann::json j = state.to_json();
        
        std::string filepath = checkpoint_dir_ + "/" + name + ".json";
        
        std::ofstream file(filepath);
        if (!file.is_open()) {
            return false;
        }
        
        file << j.dump(2);
        file.close();
        
        return true;
    }
    
    template<typename T>
    bool load_checkpoint(const std::string& name, T& state) {
        std::string filepath = checkpoint_dir_ + "/" + name + ".json";
        
        std::ifstream file(filepath);
        if (!file.is_open()) {
            return false;
        }
        
        nlohmann::json j;
        file >> j;
        
        state.from_json(j);
        
        return true;
    }
};

### 預寫日誌 (WAL)

```cpp
class WriteAheadLog {
    std::ofstream log_file_;
    std::mutex mutex_;
    
public:
    explicit WriteAheadLog(const std::string& filepath) {
        log_file_.open(filepath, std::ios::app | std::ios::binary);
    }
    
    ~WriteAheadLog() {
        if (log_file_.is_open()) {
            log_file_.close();
        }
    }
    
    void write_entry(const std::string& entry) {
        std::lock_guard<std::mutex> lock(mutex_);
        
        uint32_t length = entry.size();
        log_file_.write(reinterpret_cast<const char*>(&length), sizeof(length));
        log_file_.write(entry.data(), length);
        log_file_.flush();  // 確保寫入磁碟
    }
    
    std::vector<std::string> read_all() {
        std::vector<std::string> entries;
        
        std::ifstream file(log_file_.rdbuf());
        file.seekg(0, std::ios::beg);
        
        while (file.good()) {
            uint32_t length;
            file.read(reinterpret_cast<char*>(&length), sizeof(length));
            
            if (!file.good()) break;
            
            std::string entry(length, '\0');
            file.read(&entry[0], length);
            
            entries.push_back(std::move(entry));
        }
        
        return entries;
    }
};
```

## 斷線重連

### 連接管理器

```cpp
class ConnectionManager {
    std::string host_;
    uint16_t port_;
    int socket_fd_{-1};
    
    std::atomic<bool> connected_{false};
    std::thread reconnect_thread_;
    
    int max_reconnect_attempts_{10};
    int reconnect_delay_ms_{1000};
    
public:
    ConnectionManager(std::string host, uint16_t port)
        : host_(std::move(host)), port_(port) {}
    
    bool connect() {
        socket_fd_ = socket(AF_INET, SOCK_STREAM, 0);
        if (socket_fd_ < 0) {
            return false;
        }
        
        sockaddr_in addr;
        addr.sin_family = AF_INET;
        addr.sin_port = htons(port_);
        inet_pton(AF_INET, host_.c_str(), &addr.sin_addr);
        
        if (::connect(socket_fd_, (sockaddr*)&addr, sizeof(addr)) < 0) {
            close(socket_fd_);
            return false;
        }
        
        connected_.store(true);
        return true;
    }
    
    void start_auto_reconnect() {
        reconnect_thread_ = std::thread(&ConnectionManager::reconnect_loop, this);
    }
    
    bool is_connected() const {
        return connected_.load();
    }
    
private:
    void reconnect_loop() {
        while (true) {
            if (!connected_.load()) {
                spdlog::warn("Connection lost, attempting to reconnect...");
                
                for (int attempt = 1; attempt <= max_reconnect_attempts_; ++attempt) {
                    if (connect()) {
                        spdlog::info("Reconnected successfully");
                        on_reconnect();
                        break;
                    }
                    
                    spdlog::warn("Reconnect attempt {} failed", attempt);
                    std::this_thread::sleep_for(
                        std::chrono::milliseconds(reconnect_delay_ms_)
                    );
                }
            }
            
            std::this_thread::sleep_for(std::chrono::seconds(1));
        }
    }
    
    void on_reconnect() {
        // 重新登入、重新訂閱等
    }
};
```

## 心跳監控

### 心跳檢測器

```cpp
class HeartbeatMonitor {
    std::chrono::steady_clock::time_point last_heartbeat_;
    int timeout_seconds_;
    
    std::function<void()> on_timeout_callback_;
    
public:
    explicit HeartbeatMonitor(int timeout_sec) 
        : timeout_seconds_(timeout_sec) {
        reset();
    }
    
    void reset() {
        last_heartbeat_ = std::chrono::steady_clock::now();
    }
    
    bool check() {
        auto now = std::chrono::steady_clock::now();
        auto elapsed = std::chrono::duration_cast<std::chrono::seconds>(
            now - last_heartbeat_
        );
        
        if (elapsed.count() > timeout_seconds_) {
            if (on_timeout_callback_) {
                on_timeout_callback_();
            }
            return false;
        }
        
        return true;
    }
    
    void set_timeout_callback(std::function<void()> callback) {
        on_timeout_callback_ = std::move(callback);
    }
};
```

## 熔斷器模式

### Circuit Breaker

```cpp
class CircuitBreaker {
public:
    enum class State {
        CLOSED,      // 正常運行
        OPEN,        // 熔斷開啟
        HALF_OPEN    // 嘗試恢復
    };
    
private:
    State state_{State::CLOSED};
    
    int failure_threshold_{5};
    int success_threshold_{2};
    int timeout_ms_{60000};
    
    int failure_count_{0};
    int success_count_{0};
    
    std::chrono::steady_clock::time_point last_failure_time_;
    
public:
    bool allow_request() {
        if (state_ == State::OPEN) {
            auto now = std::chrono::steady_clock::now();
            auto elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(
                now - last_failure_time_
            );
            
            if (elapsed.count() > timeout_ms_) {
                state_ = State::HALF_OPEN;
                success_count_ = 0;
            } else {
                return false;  // 仍在熔斷中
            }
        }
        
        return true;
    }
    
    void record_success() {
        failure_count_ = 0;
        
        if (state_ == State::HALF_OPEN) {
            success_count_++;
            
            if (success_count_ >= success_threshold_) {
                state_ = State::CLOSED;
                spdlog::info("Circuit breaker closed");
            }
        }
    }
    
    void record_failure() {
        failure_count_++;
        last_failure_time_ = std::chrono::steady_clock::now();
        
        if (failure_count_ >= failure_threshold_) {
            state_ = State::OPEN;
            spdlog::warn("Circuit breaker opened");
        }
    }
    
    State state() const { return state_; }
};

// 使用範例
CircuitBreaker order_breaker;

bool submit_order_with_breaker(const Order& order) {
    if (!order_breaker.allow_request()) {
        spdlog::warn("Order rejected: circuit breaker open");
        return false;
    }
    
    try {
        bool success = submit_order(order);
        
        if (success) {
            order_breaker.record_success();
        } else {
            order_breaker.record_failure();
        }
        
        return success;
        
    } catch (const std::exception& e) {
        order_breaker.record_failure();
        throw;
    }
}
```

## 監控與告警

### 指標收集器

```cpp
class MetricsCollector {
    struct Metrics {
        std::atomic<uint64_t> orders_submitted{0};
        std::atomic<uint64_t> orders_filled{0};
        std::atomic<uint64_t> orders_rejected{0};
        std::atomic<uint64_t> messages_received{0};
        std::atomic<uint64_t> messages_processed{0};
        std::atomic<uint64_t> errors{0};
    };
    
    Metrics metrics_;
    
    std::vector<uint64_t> latency_samples_;
    std::mutex latency_mutex_;
    
public:
    void record_order_submitted() {
        metrics_.orders_submitted.fetch_add(1, std::memory_order_relaxed);
    }
    
    void record_order_filled() {
        metrics_.orders_filled.fetch_add(1, std::memory_order_relaxed);
    }
    
    void record_latency(uint64_t latency_ns) {
        std::lock_guard<std::mutex> lock(latency_mutex_);
        latency_samples_.push_back(latency_ns);
        
        // 保持最近 10000 個樣本
        if (latency_samples_.size() > 10000) {
            latency_samples_.erase(latency_samples_.begin());
        }
    }
    
    void print_summary() {
        std::cout << "=== Metrics Summary ===\n";
        std::cout << "Orders Submitted: " 
                  << metrics_.orders_submitted.load() << "\n";
        std::cout << "Orders Filled: " 
                  << metrics_.orders_filled.load() << "\n";
        std::cout << "Fill Rate: " 
                  << (100.0 * metrics_.orders_filled.load() / 
                      metrics_.orders_submitted.load()) << "%\n";
        
        // 延遲統計
        std::lock_guard<std::mutex> lock(latency_mutex_);
        if (!latency_samples_.empty()) {
            auto sorted = latency_samples_;
            std::sort(sorted.begin(), sorted.end());
            
            std::cout << "Latency P50: " 
                      << sorted[sorted.size() * 50 / 100] / 1000.0 << " us\n";
            std::cout << "Latency P99: " 
                      << sorted[sorted.size() * 99 / 100] / 1000.0 << " us\n";
        }
    }
};
```

## 災難恢復

### 備份與恢復

```cpp
class DisasterRecovery {
    std::string primary_host_;
    std::string backup_host_;
    uint16_t port_;
    
    bool using_backup_{false};
    
public:
    DisasterRecovery(std::string primary, std::string backup, uint16_t port)
        : primary_host_(std::move(primary)), 
          backup_host_(std::move(backup)),
          port_(port) {}
    
    std::string get_active_host() const {
        return using_backup_ ? backup_host_ : primary_host_;
    }
    
    void failover() {
        spdlog::warn("Failing over to backup host");
        using_backup_ = true;
    }
    
    void failback() {
        spdlog::info("Failing back to primary host");
        using_backup_ = false;
    }
};
```
