# 低延遲設計模式

## 概述

低延遲設計是高頻交易 (HFT) 系統的核心要求。本文涵蓋經過實戰驗證的設計模式，目標是將端到端延遲控制在微秒級別。

### 延遲目標

| 系統類型 | 延遲目標 | 應用場景 |
|---------|---------|---------|
| Ultra-Low | < 1 μs | Market Making, Arbitrage |
| Low | 1-10 μs | 算法交易, Smart Order Routing |
| Medium | 10-100 μs | 風險管理, 一般交易 |

### 核心原則

1. **零分配**: 避免關鍵路徑上的堆分配
2. **預分配**: 啟動時預留所有資源
3. **無鎖設計**: 使用無鎖數據結構
4. **快取友善**: 優化數據佈局
5. **系統調優**: CPU 親和性、Huge Pages

## 記憶體池模式

### 對象池實現

```cpp
#include <array>
#include <atomic>
#include <cstdint>

template<typename T, size_t Capacity>
class ObjectPool {
    struct Node {
        alignas(T) uint8_t storage[sizeof(T)];
        std::atomic<bool> in_use{false};
    };
    
    std::array<Node, Capacity> pool_;
    std::atomic<size_t> hint_{0};
    
public:
    ObjectPool() = default;
    
    // 禁止拷貝
    ObjectPool(const ObjectPool&) = delete;
    ObjectPool& operator=(const ObjectPool&) = delete;
    
    template<typename... Args>
    T* acquire(Args&&... args) {
        size_t start = hint_.load(std::memory_order_relaxed);
        
        for (size_t i = 0; i < Capacity; ++i) {
            size_t index = (start + i) % Capacity;
            
            bool expected = false;
            if (pool_[index].in_use.compare_exchange_strong(
                expected, true, std::memory_order_acquire)) {
                
                hint_.store((index + 1) % Capacity, std::memory_order_relaxed);
                
                T* ptr = reinterpret_cast<T*>(pool_[index].storage);
                new (ptr) T(std::forward<Args>(args)...);
                
                return ptr;
            }
        }
        
        return nullptr;  // 池已滿
    }
    
    void release(T* ptr) {
        if (!ptr) return;
        
        ptr->~T();
        
        size_t index = (reinterpret_cast<uint8_t*>(ptr) - 
                       reinterpret_cast<uint8_t*>(pool_.data())) / sizeof(Node);
        
        pool_[index].in_use.store(false, std::memory_order_release);
    }
};

// 使用範例
struct Order {
    uint64_t id;
    uint32_t symbol_id;
    double price;
    uint64_t quantity;
    
    Order(uint64_t i, uint32_t s, double p, uint64_t q)
        : id(i), symbol_id(s), price(p), quantity(q) {}
};

ObjectPool<Order, 10000> order_pool;

void process_order() {
    Order* order = order_pool.acquire(12345, 1001, 150.5, 100);
    
    if (order) {
        // 處理訂單...
        
        order_pool.release(order);
    }
}
```

### 固定大小記憶體池

```cpp
#include <cstdint>
#include <cstring>
#include <new>

class FixedMemoryPool {
    uint8_t* buffer_;
    size_t block_size_;
    size_t block_count_;
    uint8_t* free_list_;
    
public:
    FixedMemoryPool(size_t block_size, size_t block_count)
        : block_size_(block_size), block_count_(block_count) {
        
        // 分配連續記憶體
        buffer_ = static_cast<uint8_t*>(
            aligned_alloc(64, block_size_ * block_count_)
        );
        
        if (!buffer_) {
            throw std::bad_alloc();
        }
        
        // 初始化自由鏈表
        free_list_ = buffer_;
        
        for (size_t i = 0; i < block_count_ - 1; ++i) {
            uint8_t* current = buffer_ + i * block_size_;
            uint8_t* next = buffer_ + (i + 1) * block_size_;
            
            std::memcpy(current, &next, sizeof(uint8_t*));
        }
        
        // 最後一塊指向 null
        uint8_t* last = buffer_ + (block_count_ - 1) * block_size_;
        uint8_t* null_ptr = nullptr;
        std::memcpy(last, &null_ptr, sizeof(uint8_t*));
    }
    
    ~FixedMemoryPool() {
        free(buffer_);
    }
    
    void* allocate() {
        if (!free_list_) {
            return nullptr;
        }
        
        void* ptr = free_list_;
        
        // 更新自由鏈表
        std::memcpy(&free_list_, free_list_, sizeof(uint8_t*));
        
        return ptr;
    }
    
    void deallocate(void* ptr) {
        if (!ptr) return;
        
        // 插回自由鏈表頭部
        std::memcpy(ptr, &free_list_, sizeof(uint8_t*));
        free_list_ = static_cast<uint8_t*>(ptr);
    }
};

// 使用範例
FixedMemoryPool message_pool(256, 10000);

void* allocate_message() {
    return message_pool.allocate();
}

void free_message(void* msg) {
    message_pool.deallocate(msg);
}
```

## 無鎖環形緩衝區

### 單生產者單消費者 (SPSC)

```cpp
#include <atomic>
#include <array>
#include <cstdint>

template<typename T, size_t Capacity>
class SPSCQueue {
    static_assert((Capacity & (Capacity - 1)) == 0, "Capacity must be power of 2");
    
    std::array<T, Capacity> buffer_;
    alignas(64) std::atomic<size_t> head_{0};
    alignas(64) std::atomic<size_t> tail_{0};
    
    static constexpr size_t mask = Capacity - 1;
    
public:
    bool push(const T& item) {
        size_t current_tail = tail_.load(std::memory_order_relaxed);
        size_t next_tail = (current_tail + 1) & mask;
        
        if (next_tail == head_.load(std::memory_order_acquire)) {
            return false;  // 佇列滿
        }
        
        buffer_[current_tail] = item;
        
        tail_.store(next_tail, std::memory_order_release);
        
        return true;
    }
    
    bool pop(T& item) {
        size_t current_head = head_.load(std::memory_order_relaxed);
        
        if (current_head == tail_.load(std::memory_order_acquire)) {
            return false;  // 佇列空
        }
        
        item = buffer_[current_head];
        
        size_t next_head = (current_head + 1) & mask;
        head_.store(next_head, std::memory_order_release);
        
        return true;
    }
    
    size_t size() const {
        size_t current_head = head_.load(std::memory_order_acquire);
        size_t current_tail = tail_.load(std::memory_order_acquire);
        
        return (current_tail - current_head) & mask;
    }
    
    bool empty() const {
        return head_.load(std::memory_order_acquire) == 
               tail_.load(std::memory_order_acquire);
    }
};

// 使用範例
struct MarketTick {
    uint64_t timestamp;
    uint32_t symbol_id;
    double price;
    uint64_t volume;
};

SPSCQueue<MarketTick, 4096> tick_queue;

// 生產者執行緒
void producer_thread() {
    MarketTick tick{get_timestamp(), 1001, 150.5, 100};
    
    while (!tick_queue.push(tick)) {
        // 自旋等待
        _mm_pause();
    }
}

// 消費者執行緒
void consumer_thread() {
    MarketTick tick;
    
    while (!tick_queue.pop(tick)) {
        _mm_pause();
    }
    
    // 處理 tick...
}
```

### 批次操作優化

```cpp
template<typename T, size_t Capacity>
class BatchSPSCQueue : public SPSCQueue<T, Capacity> {
public:
    // 批次寫入
    size_t push_batch(const T* items, size_t count) {
        size_t current_tail = this->tail_.load(std::memory_order_relaxed);
        size_t current_head = this->head_.load(std::memory_order_acquire);
        
        size_t available = (Capacity - 1 - (current_tail - current_head)) & this->mask;
        size_t to_push = std::min(count, available);
        
        for (size_t i = 0; i < to_push; ++i) {
            this->buffer_[(current_tail + i) & this->mask] = items[i];
        }
        
        size_t new_tail = (current_tail + to_push) & this->mask;
        this->tail_.store(new_tail, std::memory_order_release);
        
        return to_push;
    }
    
    // 批次讀取
    size_t pop_batch(T* items, size_t count) {
        size_t current_head = this->head_.load(std::memory_order_relaxed);
        size_t current_tail = this->tail_.load(std::memory_order_acquire);
        
        size_t available = (current_tail - current_head) & this->mask;
        size_t to_pop = std::min(count, available);
        
        for (size_t i = 0; i < to_pop; ++i) {
            items[i] = this->buffer_[(current_head + i) & this->mask];
        }
        
        size_t new_head = (current_head + to_pop) & this->mask;
        this->head_.store(new_head, std::memory_order_release);
        
        return to_pop;
    }
};
```

## 零拷貝設計

### 共享記憶體訊息傳遞

```cpp
#include <sys/mman.h>
#include <fcntl.h>
#include <unistd.h>
#include <cstring>
#include <string>

class SharedMemoryRegion {
    void* ptr_;
    size_t size_;
    int fd_;
    std::string name_;
    
public:
    SharedMemoryRegion(const std::string& name, size_t size, bool create)
        : size_(size), name_(name) {
        
        int flags = create ? (O_CREAT | O_RDWR) : O_RDWR;
        
        fd_ = shm_open(name_.c_str(), flags, 0666);
        if (fd_ == -1) {
            throw std::runtime_error("shm_open failed");
        }
        
        if (create) {
            if (ftruncate(fd_, size_) == -1) {
                close(fd_);
                throw std::runtime_error("ftruncate failed");
            }
        }
        
        ptr_ = mmap(nullptr, size_, PROT_READ | PROT_WRITE, MAP_SHARED, fd_, 0);
        
        if (ptr_ == MAP_FAILED) {
            close(fd_);
            throw std::runtime_error("mmap failed");
        }
    }
    
    ~SharedMemoryRegion() {
        if (ptr_ != MAP_FAILED) {
            munmap(ptr_, size_);
        }
        
        if (fd_ != -1) {
            close(fd_);
        }
    }
    
    void* data() { return ptr_; }
    size_t size() const { return size_; }
};

// 零拷貝訊息佇列
template<typename T>
class ZeroCopyQueue {
    struct Header {
        std::atomic<uint64_t> write_seq{0};
        std::atomic<uint64_t> read_seq{0};
        char padding[64 - 2 * sizeof(std::atomic<uint64_t>)];
    };
    
    Header* header_;
    T* buffer_;
    size_t capacity_;
    
public:
    ZeroCopyQueue(SharedMemoryRegion& shm, size_t capacity)
        : capacity_(capacity) {
        
        header_ = static_cast<Header*>(shm.data());
        buffer_ = reinterpret_cast<T*>(header_ + 1);
    }
    
    T* acquire_write_slot() {
        uint64_t write_seq = header_->write_seq.load(std::memory_order_relaxed);
        uint64_t read_seq = header_->read_seq.load(std::memory_order_acquire);
        
        if (write_seq - read_seq >= capacity_) {
            return nullptr;  // 佇列滿
        }
        
        return &buffer_[write_seq % capacity_];
    }
    
    void commit_write() {
        header_->write_seq.fetch_add(1, std::memory_order_release);
    }
    
    const T* acquire_read_slot() {
        uint64_t read_seq = header_->read_seq.load(std::memory_order_relaxed);
        uint64_t write_seq = header_->write_seq.load(std::memory_order_acquire);
        
        if (read_seq >= write_seq) {
            return nullptr;  // 佇列空
        }
        
        return &buffer_[read_seq % capacity_];
    }
    
    void commit_read() {
        header_->read_seq.fetch_add(1, std::memory_order_release);
    }
};

// 使用範例
struct MarketData {
    uint64_t timestamp;
    uint32_t symbol_id;
    double bid;
    double ask;
};

void zero_copy_example() {
    SharedMemoryRegion shm("/market_data", 1024 * 1024, true);
    ZeroCopyQueue<MarketData> queue(shm, 1000);
    
    // 寫入者
    if (auto* slot = queue.acquire_write_slot()) {
        slot->timestamp = get_timestamp();
        slot->symbol_id = 1001;
        slot->bid = 150.50;
        slot->ask = 150.51;
        
        queue.commit_write();
    }
    
    // 讀取者
    if (const auto* slot = queue.acquire_read_slot()) {
        // 直接訪問，無拷貝
        process_market_data(*slot);
        
        queue.commit_read();
    }
}
```

## Disruptor 模式

### 簡化 Disruptor 實現

```cpp
#include <atomic>
#include <array>
#include <thread>
#include <vector>

template<typename T, size_t Size>
class RingBuffer {
    static_assert((Size & (Size - 1)) == 0, "Size must be power of 2");
    
    struct Slot {
        alignas(64) T data;
        alignas(64) std::atomic<int64_t> sequence{-1};
    };
    
    std::array<Slot, Size> buffer_;
    alignas(64) std::atomic<int64_t> cursor_{-1};
    
    static constexpr size_t mask = Size - 1;
    
public:
    // 生產者發布
    int64_t next() {
        return cursor_.fetch_add(1, std::memory_order_relaxed) + 1;
    }
    
    T* get(int64_t sequence) {
        return &buffer_[sequence & mask].data;
    }
    
    void publish(int64_t sequence) {
        buffer_[sequence & mask].sequence.store(sequence, std::memory_order_release);
    }
    
    // 消費者讀取
    int64_t get_cursor() const {
        return cursor_.load(std::memory_order_acquire);
    }
    
    bool is_available(int64_t sequence) {
        return buffer_[sequence & mask].sequence.load(std::memory_order_acquire) == sequence;
    }
};

// 序列屏障
class SequenceBarrier {
    std::vector<std::atomic<int64_t>*> sequences_;
    
public:
    void add_sequence(std::atomic<int64_t>* seq) {
        sequences_.push_back(seq);
    }
    
    int64_t wait_for(int64_t sequence) {
        int64_t available = std::numeric_limits<int64_t>::max();
        
        for (auto* seq : sequences_) {
            int64_t value = seq->load(std::memory_order_acquire);
            available = std::min(available, value);
        }
        
        return available;
    }
};

// 事件處理器
template<typename T>
class EventProcessor {
    RingBuffer<T, 4096>& ring_buffer_;
    SequenceBarrier& barrier_;
    std::atomic<int64_t> sequence_{-1};
    std::atomic<bool> running_{false};
    
public:
    EventProcessor(RingBuffer<T, 4096>& rb, SequenceBarrier& barrier)
        : ring_buffer_(rb), barrier_(barrier) {
        
        barrier_.add_sequence(&sequence_);
    }
    
    void run() {
        running_.store(true, std::memory_order_relaxed);
        
        int64_t next_sequence = sequence_.load(std::memory_order_relaxed) + 1;
        
        while (running_.load(std::memory_order_relaxed)) {
            int64_t available = barrier_.wait_for(next_sequence);
            
            while (next_sequence <= available) {
                if (ring_buffer_.is_available(next_sequence)) {
                    T* event = ring_buffer_.get(next_sequence);
                    
                    // 處理事件
                    process_event(*event);
                    
                    next_sequence++;
                }
            }
            
            sequence_.store(next_sequence - 1, std::memory_order_release);
        }
    }
    
    void stop() {
        running_.store(false, std::memory_order_relaxed);
    }
    
private:
    void process_event(const T& event) {
        // 事件處理邏輯
    }
};
```

```cpp
#include <array>
#include <cmath>

class FastMath {
    static constexpr size_t TABLE_SIZE = 10000;
    
    std::array<double, TABLE_SIZE> log_table_;
    std::array<double, TABLE_SIZE> exp_table_;
    std::array<double, TABLE_SIZE> sqrt_table_;
    
public:
    FastMath() {
        // 預先計算
        for (size_t i = 0; i < TABLE_SIZE; ++i) {
            double x = static_cast<double>(i) / 100.0;
            
            log_table_[i] = std::log(x + 1.0);
            exp_table_[i] = std::exp(x);
            sqrt_table_[i] = std::sqrt(x);
        }
    }
    
    double fast_log(double x) const {
        if (x <= 0.0 || x > 100.0) {
            return std::log(x);  // 回退到精確計算
        }
        
        size_t index = static_cast<size_t>(x * 100.0);
        return log_table_[index];
    }
    
    double fast_exp(double x) const {
        if (x < 0.0 || x > 100.0) {
            return std::exp(x);
        }
        
        size_t index = static_cast<size_t>(x * 100.0);
        return exp_table_[index];
    }
    
    double fast_sqrt(double x) const {
        if (x < 0.0 || x > 100.0) {
            return std::sqrt(x);
        }
        
        size_t index = static_cast<size_t>(x * 100.0);
        return sqrt_table_[index];
    }
};

// 快速價格轉換
class PriceConverter {
    static constexpr size_t MAX_PRICE_TICKS = 100000;
    std::array<uint64_t, MAX_PRICE_TICKS> price_to_ticks_;
    std::array<double, MAX_PRICE_TICKS> ticks_to_price_;
    
    double tick_size_;
    
public:
    explicit PriceConverter(double tick_size) : tick_size_(tick_size) {
        for (size_t i = 0; i < MAX_PRICE_TICKS; ++i) {
            ticks_to_price_[i] = i * tick_size_;
            price_to_ticks_[i] = i;
        }
    }
    
    uint64_t price_to_ticks(double price) const {
        size_t index = static_cast<size_t>(price / tick_size_);
        
        if (index < MAX_PRICE_TICKS) {
            return price_to_ticks_[index];
        }
        
        return static_cast<uint64_t>(price / tick_size_);
    }
    
    double ticks_to_price(uint64_t ticks) const {
        if (ticks < MAX_PRICE_TICKS) {
            return ticks_to_price_[ticks];
        }
        
        return ticks * tick_size_;
    }
};
```

## 分支消除技巧

### 無分支選擇

```cpp
#include <cstdint>
#include <algorithm>

// 有分支 (慢)
int max_with_branch(int a, int b) {
    if (a > b) {
        return a;
    } else {
        return b;
    }
}

// 無分支 (快)
int max_without_branch(int a, int b) {
    int diff = a - b;
    int sign = (diff >> 31) & 1;  // 取符號位
    return a - (diff & -sign);
}

// 或使用標準庫
int max_std(int a, int b) {
    return std::max(a, b);  // 編譯器會優化
}

// 條件賦值
void conditional_update_with_branch(int& value, int new_value, bool condition) {
    if (condition) {
        value = new_value;
    }
}

void conditional_update_without_branch(int& value, int new_value, bool condition) {
    value = condition ? new_value : value;
    // 或
    value = new_value * condition + value * !condition;
}
```

## 系統調優模式

### CPU 親和性設定

```cpp
#include <pthread.h>
#include <sched.h>
#include <thread>

class CPUAffinity {
public:
    static bool set_affinity(std::thread& thread, int cpu_id) {
        cpu_set_t cpuset;
        CPU_ZERO(&cpuset);
        CPU_SET(cpu_id, &cpuset);
        
        int result = pthread_setaffinity_np(
            thread.native_handle(),
            sizeof(cpu_set_t),
            &cpuset
        );
        
        return result == 0;
    }
    
    static bool set_current_thread_affinity(int cpu_id) {
        cpu_set_t cpuset;
        CPU_ZERO(&cpuset);
        CPU_SET(cpu_id, &cpuset);
        
        return sched_setaffinity(0, sizeof(cpu_set_t), &cpuset) == 0;
    }
};

void critical_thread_func() {
    CPUAffinity::set_current_thread_affinity(2);  // 綁定到 CPU 2

}
```

## 完整範例: 低延遲訊息處理器

```cpp
#include <atomic>
#include <thread>
#include <cstdint>

// 訊息定義
struct Message {
    uint64_t timestamp;
    uint32_t type;
    uint32_t symbol_id;
    double price;
    uint64_t quantity;
};

// 低延遲訊息處理器
class LowLatencyMessageProcessor {
    // 無鎖佇列
    SPSCQueue<Message, 4096> input_queue_;
    
    // 對象池
    ObjectPool<Message, 10000> message_pool_;
    
    // 處理執行緒
    std::thread processor_thread_;
    std::atomic<bool> running_{false};
    
    // 統計
    alignas(64) std::atomic<uint64_t> messages_processed_{0};
    alignas(64) std::atomic<uint64_t> total_latency_ns_{0};
    
public:
    LowLatencyMessageProcessor() {
        processor_thread_ = std::thread(&LowLatencyMessageProcessor::process_loop, this);
        
        // 設定 CPU 親和性
        CPUAffinity::set_affinity(processor_thread_, 2);
        
        // 設定實時優先級 (需要權限)
        RealtimeScheduler::set_realtime_priority(processor_thread_, 80);
    }
    
    ~LowLatencyMessageProcessor() {
        running_.store(false, std::memory_order_relaxed);
        if (processor_thread_.joinable()) {
            processor_thread_.join();
        }
    }
    
    bool enqueue(const Message& msg) {
        return input_queue_.push(msg);
    }
    
    void print_stats() const {
        uint64_t count = messages_processed_.load(std::memory_order_relaxed);
        uint64_t total_latency = total_latency_ns_.load(std::memory_order_relaxed);
        
        if (count > 0) {
            double avg_latency_us = (total_latency / count) / 1000.0;
            std::cout << "Processed: " << count 
                      << ", Avg latency: " << avg_latency_us << " us\n";
        }
    }
    
private:
    void process_loop() {
        running_.store(true, std::memory_order_relaxed);
        
        Message msg;
        
        while (running_.load(std::memory_order_relaxed)) {
            if (input_queue_.pop(msg)) {
                uint64_t start = rdtsc();
                
                // 處理訊息
                process_message(msg);
                
                uint64_t end = rdtsc();
                uint64_t latency = end - start;
                
                messages_processed_.fetch_add(1, std::memory_order_relaxed);
                total_latency_ns_.fetch_add(latency, std::memory_order_relaxed);
            } else {
                _mm_pause();
            }
        }
    }
    
    void process_message(const Message& msg) {
        // 實際處理邏輯
        // 確保無堆分配、無鎖、無分支
    }
    
    static inline uint64_t rdtsc() {
        uint32_t lo, hi;
        __asm__ volatile("rdtsc" : "=a"(lo), "=d"(hi));
        return (static_cast<uint64_t>(hi) << 32) | lo;
    }
};
```

### RDTSC 計時

```cpp
#include <cstdint>
#include <x86intrin.h>

class LatencyMeasurement {
    uint64_t start_tsc_;
    
public:
    void start() {
        _mm_lfence();  // 記憶體柵欄
        start_tsc_ = __rdtsc();
    }
    
    uint64_t end_cycles() {
        uint64_t end_tsc = __rdtsc();
        _mm_lfence();
        
        return end_tsc - start_tsc_;
    }
    
    static uint64_t cycles_to_ns(uint64_t cycles, uint64_t cpu_freq_mhz) {
        return (cycles * 1000) / cpu_freq_mhz;
    }
};

// 使用範例
void measure_operation_latency() {
    LatencyMeasurement lat;
    
    lat.start();
    
    // 被測操作
    perform_critical_operation();
    
    uint64_t cycles = lat.end_cycles();
    uint64_t ns = LatencyMeasurement::cycles_to_ns(cycles, 3600);  // 3.6 GHz
    
    std::cout << "Latency: " << cycles << " cycles (" << ns << " ns)\n";
}
```
