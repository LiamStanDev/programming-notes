## 1. 異常基礎

### 1.1 異常機制

```cpp
// 拋出異常
void divide(int a, int b) {
    if (b == 0) {
        throw std::invalid_argument("Division by zero");
    }
    return a / b;
}

// 捕獲異常
try {
    int result = divide(10, 0);
} catch (const std::invalid_argument& e) {
    std::cerr << "Error: " << e.what() << "\n";
} catch (const std::exception& e) {
    std::cerr << "Unexpected error: " << e.what() << "\n";
} catch (...) {
    std::cerr << "Unknown error\n";
}
```

### 1.2 標準異常層次

```cpp
// std::exception 層次結構
// std::exception          
//   ├── std::logic_error      (內部邏輯檢查: 程式設計師的錯)
//   │   ├── std::invalid_argument
//   │   ├── std::domain_error      
//   │   ├── std::length_error      
//   │   └── std::out_of_range
//   ├── std::runtime_error    (外部環境/計算問題: 不可預見的錯)
//   │    ├── std::range_error
//   │    ├── std::overflow_error
//   │    └── std::underflow_error
//   └── std::system_error (系統錯誤: 包裝 error_code，處理 OS/API 錯誤)

// std::system_error 詳解
void demonstrate_system_error() {
    try {
        // system_error 包裝了 error_code
        throw std::system_error(ENOENT, std::system_category(), "File operation failed");
    } catch (const std::system_error& e) {
        std::cerr << "What: " << e.what() << "\n";                    // 完整訊息
        std::cerr << "Code: " << e.code().value() << "\n";             // 錯誤碼
        std::cerr << "Message: " << e.code().message() << "\n";        // 錯誤訊息
        std::cerr << "Category: " << e.code().category().name() << "\n"; // 錯誤分類
    } catch (const std::runtime_error& e) {
        // system_error 也可以用父類捕獲
        std::cerr << "Runtime error: " << e.what() << "\n";
    }
}

// 使用
void process(int value) {
    if (value < 0) {
        throw std::invalid_argument("Value must be non-negative");
    }
    if (value > MAX_VALUE) {
        throw std::out_of_range("Value exceeds maximum");
    }
}
```

### 1.3 自定義異常

```cpp
class OrderException : public std::runtime_error {
    int order_id_;

public:

        : std::runtime_error(msg), order_id_(id) {}
```

## 2. 系統錯誤處理

C++ 提供了多種處理系統級錯誤的機制，從傳統的 C 風格錯誤處理到現代的 C++11 錯誤處理框架。

### 2.1 傳統 C 錯誤處理

#### errno 與 strerror

```cpp
#include <cerrno>
#include <cstring>
#include <iostream>
#include <fcntl.h>
#include <unistd.h>

// 傳統 C 風格錯誤處理
void open_file_c_style(const char* filename) {
    int fd = open(filename, O_RDONLY);
    if (fd == -1) {
        // errno 是全域變數，由系統調用設置
        std::cerr << "Error opening " << filename << ": " 
                  << strerror(errno) << " (errno=" << errno << ")\n";
        return;
    }
    
    // 使用檔案描述符...
    close(fd);
}

// perror 提供更簡便的錯誤輸出
void open_file_with_perror(const char* filename) {
    int fd = open(filename, O_RDONLY);
    if (fd == -1) {
        perror("open");  // 自動輸出 "open: <error message>"
        return;
    }
    
    close(fd);
}
```

#### 線程安全問題

```cpp
// ❌ 不安全：errno 在多線程中可能被覆蓋
void unsafe_error_handling() {
    int fd = open("file.txt", O_RDONLY);
    if (fd == -1) {
        // 在這裡其他線程可能已經修改了 errno！
        std::cerr << "Error: " << strerror(errno) << "\n";
    }
}

// ✅ 安全：立即保存 errno
void safe_error_handling() {
    int fd = open("file.txt", O_RDONLY);
    if (fd == -1) {
        int saved_errno = errno;  // 立即保存
        std::cerr << "Error: " << strerror(saved_errno) << "\n";
    }
}
```

### 2.2 C++11 系統錯誤框架

#### std::error_code 基礎

```cpp
#include <iostream>

// std::error_code 封裝錯誤碼和錯誤分類
void demonstrate_error_code() {

    if (ec) {
        std::cout << "Error occurred\n";
    }
}
```

#### std::error_category

```cpp
// 系統提供的兩個主要錯誤分類
void demonstrate_error_categories() {
    // 1. system_category: 平台特定的系統錯誤 (如 errno)
    auto sys_ec = std::error_code(EACCES, std::system_category());
    std::cout << "System error: " << sys_ec.message() << "\n";
    
    // 2. generic_category: POSIX 可移植錯誤碼
    auto gen_ec = std::error_code(static_cast<int>(std::errc::permission_denied),

    std::cout << "Are equal: " << (sys_ec == gen_ec) << "\n";
}
```

#### 實用的錯誤檢查函數

```cpp
#include <fstream>

// 返回 error_code 的檔案操作
std::error_code safe_read_file(const std::string& filename, std::string& content) {
    std::ifstream file(filename);
    if (!file) {
        // 將 errno 轉換為 error_code
        return std::error_code(errno, std::system_category());
    }
    
    // 讀取檔案內容
    content.assign(std::istreambuf_iterator<char>(file),
                   std::istreambuf_iterator<char>());
    
    if (file.bad()) {
        return std::error_code(EIO, std::system_category());
    }
    
    return {};  // 成功，返回默認構造的 error_code (無錯誤)
}

// 使用
void use_safe_read_file() {
    std::string content;
    auto ec = safe_read_file("data.txt", content);
    
    if (ec) {
        std::cerr << "Failed to read file: " << ec.message() << "\n";
        
        // 可以檢查特定錯誤類型
        if (ec == std::errc::no_such_file_or_directory) {
            std::cerr << "File does not exist\n";
        } else if (ec == std::errc::permission_denied) {
            std::cerr << "Permission denied\n";
        }
    } else {
        std::cout << "File read successfully, size: " << content.size() << "\n";
    }
}
```

### 2.3 std::system_error 異常

```cpp
#include <system_error>

// std::system_error 是包裝 error_code 的異常類別
void throw_system_error() {
    int fd = open("nonexistent.txt", O_RDONLY);
    if (fd == -1) {
        // 拋出系統錯誤異常
        throw std::system_error(errno, std::system_category(), 
                               "Failed to open nonexistent.txt");
    }
}

// 捕獲和處理 system_error
void handle_system_error() {
    try {
        throw_system_error();
    } catch (const std::system_error& e) {
        std::cerr << "System error: " << e.what() << "\n";
        std::cerr << "Error code: " << e.code().value() << "\n";
        std::cerr << "Error message: " << e.code().message() << "\n";
        
        // 檢查特定錯誤
        if (e.code() == std::errc::no_such_file_or_directory) {
            std::cerr << "The file does not exist\n";
        }
    }
}

// RAII 風格的檔案操作類別
class FileDescriptor {
    int fd_;

public:
    explicit FileDescriptor(const std::string& filename, int flags) {
        fd_ = open(filename.c_str(), flags);
        if (fd_ == -1) {
            throw std::system_error(errno, std::system_category(), 
                                   "Failed to open " + filename);
        }
    }
    
    ~FileDescriptor() {
        if (fd_ != -1) {
            close(fd_);
        }
    }
    
    FileDescriptor(const FileDescriptor&) = delete;
    FileDescriptor& operator=(const FileDescriptor&) = delete;
    
    FileDescriptor(FileDescriptor&& other) noexcept : fd_(other.fd_) {
        other.fd_ = -1;
    }
    
    FileDescriptor& operator=(FileDescriptor&& other) noexcept {
        if (this != &other) {
            if (fd_ != -1) close(fd_);
            fd_ = other.fd_;
            other.fd_ = -1;
        }
        return *this;
    }
    
    int get() const { return fd_; }
};
```

### 2.4 自定義錯誤分類

```cpp
// 自定義錯誤列舉
enum class NetworkError {
    SUCCESS = 0,
    CONNECTION_FAILED,
    TIMEOUT,
    INVALID_ADDRESS,
    PROTOCOL_ERROR
};

// 自定義錯誤分類
class NetworkErrorCategory : public std::error_category {
public:
    const char* name() const noexcept override {
        return "network";
    }
    
    std::string message(int ev) const override {
        switch (static_cast<NetworkError>(ev)) {
        case NetworkError::SUCCESS:
            return "Success";
        case NetworkError::CONNECTION_FAILED:
            return "Connection failed";
        case NetworkError::TIMEOUT:
            return "Operation timeout";
        case NetworkError::INVALID_ADDRESS:
            return "Invalid network address";
        case NetworkError::PROTOCOL_ERROR:
            return "Protocol error";
        default:
            return "Unknown network error";
        }
    }
    
    // 可選：提供與標準錯誤碼的比較
    bool equivalent(int code, const std::error_condition& condition) const noexcept override {
        switch (static_cast<NetworkError>(code)) {
        case NetworkError::CONNECTION_FAILED:
            return condition == std::errc::connection_refused;
        case NetworkError::TIMEOUT:
            return condition == std::errc::timed_out;
        case NetworkError::INVALID_ADDRESS:
            return condition == std::errc::invalid_argument;
        default:
            return false;
        }
    }
};

// 全域錯誤分類實例
const NetworkErrorCategory& network_category() {
    static NetworkErrorCategory instance;
    return instance;
}

// 便利函數：創建網路錯誤碼
std::error_code make_error_code(NetworkError e) {
    return std::error_code(static_cast<int>(e), network_category());
}

// 讓 NetworkError 可以自動轉換為 error_code
namespace std {
    template<>
    struct is_error_code_enum<NetworkError> : true_type {};
}

// 使用自定義錯誤
class NetworkConnection {
public:
    std::error_code connect(const std::string& host, int port) {
        // 模擬網路連線
        if (host.empty()) {
            return NetworkError::INVALID_ADDRESS;
        }
        
        if (port <= 0 || port > 65535) {
            return NetworkError::INVALID_ADDRESS;
        }
        
        // 模擬連線失敗
        if (host == "unreachable.com") {
            return NetworkError::CONNECTION_FAILED;
        }
        
        return NetworkError::SUCCESS;
    }
    
    void connect_with_exception(const std::string& host, int port) {
        auto ec = connect(host, port);
        if (ec) {
            throw std::system_error(ec, "Failed to connect to " + host);
        }
    }
};

// 使用範例
void network_example() {
    NetworkConnection conn;
    
    // 使用錯誤碼
    auto ec = conn.connect("unreachable.com", 80);
    if (ec) {
        std::cerr << "Connection failed: " << ec.message() << "\n";
        
        // 可以與標準錯誤條件比較
        if (ec == std::errc::connection_refused) {
            std::cerr << "Connection was refused\n";
        }
    }
    
    // 使用異常
    try {
        conn.connect_with_exception("", 80);
    } catch (const std::system_error& e) {
        std::cerr << "Network error: " << e.what() << "\n";
    }
}
```

## 3. 異常安全保證

### 2.1 三種安全保證

**1. 基本保證 (Basic Guarantee)**:

- 不洩漏資源
- 對象處於有效狀態 (但未指定)

**2. 強保證 (Strong Guarantee)**:

- 操作成功或完全回滾
- 對象狀態不變

**3. 不拋出保證 (No-throw Guarantee)**:

- 操作保證不拋出異常
- 通常標記為 `noexcept`

```cpp
// 基本保證
void basic_guarantee(std::vector<int>& vec, int value) {
    vec.push_back(value);  // 可能拋出,但 vec 仍有效
}

// 強保證 (Copy-and-swap idiom)
class Widget {
    std::vector<int> data_;

public:
    Widget& operator=(const Widget& other) {
        Widget temp(other);  // 可能拋出
        swap(temp);          // 不拋出
        return *this;
    }

    void swap(Widget& other) noexcept {
        data_.swap(other.data_);
    }
};

// 不拋出保證
void no_throw() noexcept {
    // 保證不拋出異常
}
```

### 3.2 Copy-and-Swap Idiom

```cpp
class Buffer {
    char* data_;
    size_t size_;

public:
    // 強異常安全的賦值
    Buffer& operator=(const Buffer& other) {
        if (this != &other) {
            Buffer temp(other);  // 可能拋出
            swap(temp);          // 不拋出
        }
        return *this;
    }

    void swap(Buffer& other) noexcept {
        std::swap(data_, other.data_);
        std::swap(size_, other.size_);
    }
};
```

### 2.3 noexcept 的重要性

```cpp
class Widget {
public:
    // ✅ 移動操作應標記為 noexcept
    Widget(Widget&&) noexcept;
    Widget& operator=(Widget&&) noexcept;

    // ✅ swap 應標記為 noexcept
    void swap(Widget&) noexcept;

    // ✅ 解構函式隱式 noexcept
    ~Widget();
};

// 為什麼重要?
std::vector<Widget> vec;
vec.push_back(Widget{});

// 如果移動建構函式不是 noexcept:
// - vector 重新分配時會拷貝而非移動
// - 性能大幅下降!
```

## 3. RAII 與異常安全

### 3.1 RAII 自動清理

```cpp
// ❌ 手動清理 (不安全)
void bad_example() {
    int* p = new int(42);
    process(p);  // 可能拋出異常
    delete p;    // 不會執行!
}

// ✅ RAII 自動清理
void good_example() {
    auto p = std::make_unique<int>(42);
    process(p.get());  // 即使拋出異常,p 也會自動釋放
}
```

### 3.2 RAII 守衛

```cpp
// 文件守衛
class FileGuard {
    FILE* fp_;

public:
    explicit FileGuard(const char* filename)
        : fp_(std::fopen(filename, "r")) {
        if (!fp_) {
            throw std::runtime_error("Failed to open file");
        }
    }

    ~FileGuard() {
        if (fp_) std::fclose(fp_);
    }

    FileGuard(const FileGuard&) = delete;
    FileGuard& operator=(const FileGuard&) = delete;

    FILE* get() const { return fp_; }
};

// 使用
void process_file(const char* filename) {
    FileGuard file(filename);  // 自動管理
    // ... 處理文件
    // 即使拋出異常,文件也會自動關閉
}
```

### 3.3 鎖守衛

```cpp
std::mutex mtx;
int shared_data = 0;

// ❌ 手動加鎖 (不安全)
void bad_example() {
    mtx.lock();
    process(shared_data);  // 可能拋出異常
    mtx.unlock();          // 不會執行!
}

// ✅ RAII 鎖守衛
void good_example() {
    std::lock_guard<std::mutex> lock(mtx);
    process(shared_data);  // 即使拋出異常,鎖也會自動釋放
}

// ✅ 或使用 unique_lock (更靈活)
void flexible_example() {
    std::unique_lock<std::mutex> lock(mtx);
    process(shared_data);
    lock.unlock();  // 可以手動解鎖
    // ...
    lock.lock();    // 可以重新加鎖
}
```

## 4. 異常的替代方案

### 5.1 錯誤碼

#### 傳統錯誤碼

```cpp
enum class ErrorCode {
    SUCCESS,
    INVALID_INPUT,
    FILE_NOT_FOUND,
    NETWORK_ERROR
};

// 返回錯誤碼
ErrorCode read_file(const std::string& filename, std::string& content) {
    std::ifstream file(filename);
    if (!file) {
        return ErrorCode::FILE_NOT_FOUND;
    }

    content.assign(std::istreambuf_iterator<char>(file),
                   std::istreambuf_iterator<char>());
    return ErrorCode::SUCCESS;
}

// 使用
std::string content;
ErrorCode err = read_file("data.txt", content);
if (err != ErrorCode::SUCCESS) {
    // 處理錯誤
}
```

#### std::error_code 作為替代方案

```cpp
#include <system_error>
#include <fstream>

// 使用 std::error_code 的檔案操作
std::error_code read_file_with_error_code(const std::string& filename, std::string& content) {
    std::ifstream file(filename);
    if (!file) {
        // 直接返回系統錯誤碼
        return std::make_error_code(std::errc::no_such_file_or_directory);
    }

    content.assign(std::istreambuf_iterator<char>(file),
                   std::istreambuf_iterator<char>());
    
    if (file.bad()) {
        return std::make_error_code(std::errc::io_error);
    }
    
    return {};  // 成功 (默認構造的 error_code 表示無錯誤)
}

// 使用 std::error_code - 更豐富的錯誤訊息
void use_error_code_example() {
    std::string content;
    auto ec = read_file_with_error_code("missing.txt", content);
    
    if (ec) {
        std::cerr << "Error code: " << ec.value() << "\n";
        std::cerr << "Error message: " << ec.message() << "\n";
        std::cerr << "Error category: " << ec.category().name() << "\n";
        
        // 可以與標準錯誤條件比較
        if (ec == std::errc::no_such_file_or_directory) {
            std::cerr << "File does not exist\n";
        } else if (ec == std::errc::permission_denied) {
            std::cerr << "Access denied\n";
        }
    } else {
        std::cout << "File read successfully\n";
    }
}

// 組合多個操作的錯誤處理
std::error_code process_config_file(const std::string& filename) {
    std::string content;
    
    // 第一步：讀取檔案
    if (auto ec = read_file_with_error_code(filename, content)) {
        return ec;  // 傳播錯誤
    }
    
    // 第二步：解析內容 (假設的函數)
    if (content.empty()) {
        return std::make_error_code(std::errc::invalid_argument);
    }
    
    // 第三步：應用配置 (假設的函數)
    // ... 其他操作
    
    return {};  // 全部成功
}
```

## 5. 錯誤處理最佳實踐

### 5.1 何時使用異常?

**✅ 適合使用異常**:

- 構造函數失敗
- 資源分配失敗
- 不可恢復的錯誤
- 跨多層調用的錯誤傳播

**❌ 不適合使用異常**:

- 正常的控制流
- 高頻操作 (性能關鍵路徑)
- 可預期的錯誤 (如文件不存在)
- 嵌入式系統 (禁用異常)

```cpp
// ✅ 適合: 構造函數失敗
class Connection {
public:
    Connection(const std::string& host) {
        if (!connect(host)) {
            throw std::runtime_error("Connection failed");
        }
    }
};

// ❌ 不適合: 正常控制流
void bad_example() {
    try {
        for (int i = 0; ; ++i) {
            process(i);
        }
    } catch (const EndOfData&) {
        // 不要用異常控制循環!
    }
}
```

```cpp
class OrderBook {
    std::vector<Order> orders_;

public:
    // ✅ 強異常安全
    void add_order(const Order& order) {
        orders_.push_back(order);  // 可能拋出
        // 如果拋出,orders_ 保持不變
    }

    // ✅ 預分配避免異常
    void add_orders(const std::vector<Order>& new_orders) {
        orders_.reserve(orders_.size() + new_orders.size());
        for (const auto& order : new_orders) {
            orders_.push_back(order);  // 不會拋出 (已預分配)
        }
    }
};
```

### 6.5 系統調用錯誤處理最佳實踐

#### C/C++ 混合代碼的錯誤處理

```cpp
#include <system_error>
#include <cerrno>

// ✅ 包裝 C API 為 C++ 風格
class FileHandle {
    FILE* fp_;

public:
    static std::expected<FileHandle, std::error_code> open(const std::string& filename, const char* mode) {
        FILE* fp = std::fopen(filename.c_str(), mode);
        if (!fp) {
            return std::unexpected(std::error_code(errno, std::system_category()));
        }
        return FileHandle(fp);
    }

private:
    explicit FileHandle(FILE* fp) : fp_(fp) {}

public:
    ~FileHandle() {
        if (fp_) std::fclose(fp_);
    }

    FileHandle(const FileHandle&) = delete;
    FileHandle& operator=(const FileHandle&) = delete;

    FileHandle(FileHandle&& other) noexcept : fp_(other.fp_) {
        other.fp_ = nullptr;
    }

    FileHandle& operator=(FileHandle&& other) noexcept {
        if (this != &other) {
            if (fp_) std::fclose(fp_);
            fp_ = other.fp_;
            other.fp_ = nullptr;
        }
        return *this;
    }

    FILE* get() const { return fp_; }
};

// 使用
void use_wrapped_c_api() {
    auto file_result = FileHandle::open("data.txt", "r");
    if (!file_result) {
        auto ec = file_result.error();
        std::cerr << "Failed to open file: " << ec.message() << "\n";
        
        if (ec == std::errc::no_such_file_or_directory) {
            std::cerr << "File does not exist\n";
        } else if (ec == std::errc::permission_denied) {
            std::cerr << "Permission denied\n";
        }
        return;
    }
    
    // 安全使用檔案
    auto& file = file_result.value();
    // ... 檔案操作
}
```

#### 跨平台系統錯誤處理

```cpp
#include <system_error>
#ifdef _WIN32
#include <windows.h>
#else
#include <unistd.h>
#include <sys/stat.h>
#endif

// 跨平台的檔案操作錯誤處理
std::error_code get_file_size(const std::string& filename, size_t& size) {
#ifdef _WIN32
    HANDLE hFile = CreateFileA(filename.c_str(), GENERIC_READ, FILE_SHARE_READ,
                               nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr);
    if (hFile == INVALID_HANDLE_VALUE) {
        return std::error_code(GetLastError(), std::system_category());
    }

    LARGE_INTEGER fileSize;
    if (!GetFileSizeEx(hFile, &fileSize)) {
        auto ec = std::error_code(GetLastError(), std::system_category());
        CloseHandle(hFile);
        return ec;
    }

    size = static_cast<size_t>(fileSize.QuadPart);
    CloseHandle(hFile);
    return {};
#else
    struct stat st;
    if (stat(filename.c_str(), &st) != 0) {
        return std::error_code(errno, std::system_category());
    }
    
    size = static_cast<size_t>(st.st_size);
    return {};
#endif
}

// 使用跨平台錯誤處理
void demonstrate_cross_platform_errors() {
    size_t file_size;
    auto ec = get_file_size("test.txt", file_size);
    
    if (ec) {
        std::cerr << "Error getting file size: " << ec.message() << "\n";
        std::cerr << "Error category: " << ec.category().name() << "\n";
        
        // 統一的錯誤處理，不需要考慮平台差異
        if (ec == std::errc::no_such_file_or_directory) {
            std::cerr << "File not found\n";
        } else if (ec == std::errc::permission_denied) {
            std::cerr << "Access denied\n";
        }
    } else {
        std::cout << "File size: " << file_size << " bytes\n";
    }
}
```

#### 線程安全的錯誤處理

```cpp
#include <thread>
#include <mutex>
#include <system_error>

// ✅ 線程安全的系統調用包裝
class ThreadSafeFileOps {
    mutable std::mutex mutex_;

public:
    std::error_code safe_write_file(const std::string& filename, const std::string& content) {
        std::lock_guard<std::mutex> lock(mutex_);
        
        // 立即保存 errno，避免其他線程干擾
        errno = 0;  // 清零
        
        std::FILE* fp = std::fopen(filename.c_str(), "w");
        if (!fp) {
            int saved_errno = errno;  // 立即保存
            return std::error_code(saved_errno, std::system_category());
        }
        
        // 寫入操作
        size_t written = std::fwrite(content.data(), 1, content.size(), fp);
        if (written != content.size()) {
            int saved_errno = errno;
            std::fclose(fp);
            return std::error_code(saved_errno, std::system_category());
        }
        
        if (std::fclose(fp) != 0) {
            return std::error_code(errno, std::system_category());
        }
        
        return {};  // 成功
    }
};

// 多線程使用範例
void multithread_example() {
    ThreadSafeFileOps file_ops;
    
    auto worker = [&file_ops](int thread_id) {
        std::string filename = "output_" + std::to_string(thread_id) + ".txt";
        std::string content = "Data from thread " + std::to_string(thread_id);
        
        auto ec = file_ops.safe_write_file(filename, content);
        if (ec) {
            std::cerr << "Thread " << thread_id << " failed: " << ec.message() << "\n";
        } else {
            std::cout << "Thread " << thread_id << " succeeded\n";
        }
    };
    
    std::vector<std::thread> threads;
    for (int i = 0; i < 5; ++i) {
        threads.emplace_back(worker, i);
    }
    
    for (auto& t : threads) {
        t.join();
    }
}
```

#### 高頻交易場景的錯誤處理

```cpp
// 高性能場景：避免字串分配和格式化
class HighFreqErrorHandler {
    static constexpr size_t MAX_ERROR_MSG = 256;
    
public:
    // 返回預分配的錯誤訊息，避免動態分配
    static const char* get_error_message(std::error_code ec) {
        thread_local char buffer[MAX_ERROR_MSG];
        
        // 使用 C API 避免字串分配
        const char* msg = strerror(ec.value());
        strncpy(buffer, msg, MAX_ERROR_MSG - 1);
        buffer[MAX_ERROR_MSG - 1] = '\0';
        
        return buffer;
    }
    
    // 快速錯誤檢查，內聯優化
    static inline bool is_retriable_error(std::error_code ec) {
        return ec == std::errc::resource_temporarily_unavailable ||
               ec == std::errc::interrupted ||
               ec == std::errc::timed_out;
    }
};

// 高頻操作中的錯誤處理
std::error_code high_freq_operation() {
    // 避免異常，使用錯誤碼
    int result = some_system_call();
    if (result < 0) {
        auto ec = std::error_code(errno, std::system_category());
        
        // 快速檢查是否可重試
        if (HighFreqErrorHandler::is_retriable_error(ec)) {
            // 可以重試，不記錄錯誤
            return ec;
        }
        
        // 嚴重錯誤，可能需要記錄
        return ec;
    }
    
    return {};  // 成功
}
```
